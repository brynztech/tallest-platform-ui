    Internal state: const [isDragOver, setIsDragOver] = useState(false).
    Derived: const isBusy = status === 'uploading' || status === 'processing'.

    Render a <label> with these classes ALWAYS present:
      'flex min-h-[200px] cursor-pointer flex-col items-center justify-center gap-3 rounded-2xl border border-dashed px-6 text-center'
      'transition-[color,box-shadow,background-color] duration-150 ease-out'
    PITFALL: do NOT write transition-shadow (invalid). Use the bracketed multi-property form above.

    Then a cn() conditional state class (priority order — drag-over wins, then failed, then file-selected, then busy, else idle):
      isDragOver        -> 'border-primary bg-primary/5 ring-2 ring-primary/30'
      status==='failed' -> 'border-destructive/50 bg-destructive/5'
      file (truthy)     -> 'border-primary/40 bg-primary/5'
      isBusy            -> 'border-border/50 bg-muted/30'
      else (idle)       -> 'border-border/70 bg-card hover:bg-muted/50 hover:border-border'

    Drag handlers on the label (PITFALL 5 — preventDefault on onDragOver is mandatory):
      onDragEnter={() => !disabled && setIsDragOver(true)}
      onDragOver={(e) => e.preventDefault()}
      onDragLeave={() => setIsDragOver(false)}
      onDrop={(e) => { e.preventDefault(); setIsDragOver(false); if (disabled) return; const dropped = e.dataTransfer.files[0]; if (dropped) onChange(dropped) }}

    Icon: choose by state — AlertCircle when status==='failed' (text-destructive), FileText when a file is selected (text-primary), else UploadCloud (text-primary when isDragOver, otherwise text-muted-foreground). Icon size h-8 w-8.

    Text content:
      - Primary line (span text-sm font-medium text-foreground): if status==='failed' -> 'Upload failed — try again'; else if file -> {file.name}; else if isDragOver -> 'Drop it here'; else -> 'Drop your resume here, or click to upload'.
      - Subtext (span text-xs text-muted-foreground): if isBusy -> 'Reading your resume…'; else -> 'PDF, DOC, DOCX up to mock limit'.

    Hidden input (preserve click-to-upload):
      <input type="file" accept=".pdf,.doc,.docx" className="sr-only" disabled={disabled} onChange={(e) => onChange(e.target.files?.[0] ?? null)} />

    Export: export function ResumeDropzone(props: ResumeDropzoneProps).
