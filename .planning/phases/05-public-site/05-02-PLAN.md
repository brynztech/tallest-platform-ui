```typescript
'use client'
import { createContext, useContext, useEffect, useState } from 'react'

type CandidateAuth = {
  isLoggedIn: boolean
  candidate: { id: string; email: string } | null
  signIn: (token: string, candidate: { id: string; email: string }) => void
  signOut: () => void
}

const CandidateAuthContext = createContext<CandidateAuth>(null!)

export function CandidateAuthProvider({ children }: { children: React.ReactNode }) {
  const [candidate, setCandidate] = useState<{ id: string; email: string } | null>(null)

  useEffect(() => {
    // T-05-03: try/catch guards against prototype pollution via corrupt localStorage
    const token = localStorage.getItem('candidate_access_token')
    const stored = localStorage.getItem('candidate_data')
    if (token && stored) {
      try { setCandidate(JSON.parse(stored)) } catch { /* corrupt — ignore per T-05-03 */ }
    }
  }, [])

  function signIn(token: string, data: { id: string; email: string }) {
    localStorage.setItem('candidate_access_token', token)
    localStorage.setItem('candidate_data', JSON.stringify(data))
    setCandidate(data)
  }

  function signOut() {
    localStorage.removeItem('candidate_access_token')
    localStorage.removeItem('candidate_data')
    setCandidate(null)
  }

  return (
    <CandidateAuthContext.Provider value={{ isLoggedIn: !!candidate, candidate, signIn, signOut }}>
      {children}
    </CandidateAuthContext.Provider>
  )
}

export function useCandidateAuth() {
  return useContext(CandidateAuthContext)
}
```

```typescript
'use client'
import Link from 'next/link'
import { Badge } from '@brynzit/ui/components/ui/badge'
import { Card } from '@brynzit/ui/components/ui/card'
import { cn } from '@brynzit/ui/lib/utils'
import { MapPin } from 'lucide-react'
import type { PublicJobItem } from '../../_lib/candidate-types'

interface JobCardProps {
  job: PublicJobItem
  isLoggedIn: boolean
  isLocked?: boolean
}

function formatComp(min: number | null, max: number | null): string | null {
  if (!min) return null
  const fmt = (n: number) => `$${(n / 1000).toFixed(0)}k`
  return max && max !== min ? `${fmt(min)}–${fmt(max)}` : fmt(min)
}

const EMPLOYMENT_LABELS: Record<PublicJobItem['employment_type'], string> = {
  full_time: 'Full-time',
  part_time: 'Part-time',
  contract: 'Contract',
}

export function JobCard({ job, isLoggedIn, isLocked }: JobCardProps) {
  return (
    <Link href={`/jobs/${job.id}`} className="block" aria-hidden={isLocked}>
      <Card
        className={cn(
          'cursor-pointer p-4 transition-colors hover:bg-muted',
          isLocked && 'blur-[4px] pointer-events-none select-none',
        )}
      >
        <div className="space-y-2">
          <p className="text-base text-foreground">{job.title}</p>
          {job.company_name && (
            <p className="text-sm text-muted-foreground">{job.company_name}</p>
          )}
          {job.location && (
            <p className="flex items-center gap-1 text-sm text-muted-foreground">
              <MapPin className="h-4 w-4 shrink-0" />
              {job.location}
            </p>
          )}
          {formatComp(job.compensation_min, job.compensation_max) && (
            <p className="text-sm text-muted-foreground">
              {formatComp(job.compensation_min, job.compensation_max)}
            </p>
          )}
          <div className="flex flex-wrap gap-1 pt-1">
            <Badge variant="secondary">{EMPLOYMENT_LABELS[job.employment_type]}</Badge>
            {job.skills.slice(0, 3).map(skill => (
              <Badge key={skill} variant="outline">{skill}</Badge>
            ))}
            {job.skills.length > 3 && (
              <Badge variant="outline">+{job.skills.length - 3} more</Badge>
            )}
            {isLoggedIn && job.match_strength === 'strong' && (
              <Badge variant="default">Strong match</Badge>
            )}
            {isLoggedIn && job.match_strength === 'partial' && (
              <Badge variant="secondary">Partial match</Badge>
            )}
          </div>
        </div>
      </Card>
    </Link>
  )
}
```

```typescript
'use client'
import { useEffect, useRef, useState } from 'react'
import { useInfiniteQuery } from '@tanstack/react-query'
import { Skeleton } from '@brynzit/ui/components/ui/skeleton'
import { Button } from '@brynzit/ui/components/ui/button'
import { JobCard } from './_components/job-card'
import { FilterBar, type JobFilters } from './_components/filter-bar'
import { LockedJobsOverlay } from './_components/locked-jobs-overlay'
import { useCandidateAuth } from '../_components/candidate-auth-context'
import type { PublicJobItem, PublicJobsResponse } from '../_lib/candidate-types'

const API_BASE = process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:8000'
const VISIBLE_LIMIT = 4

export default function JobsPage() {
  const { isLoggedIn } = useCandidateAuth()
  const [searchText, setSearchText] = useState('')
  const [debouncedSearch, setDebouncedSearch] = useState('')
  const [filters, setFilters] = useState<JobFilters>({ role_type: '', seniority: '', remote: false })
  const loadMoreRef = useRef<HTMLDivElement>(null)

  // 350ms debounce for text search (D-08)
  useEffect(() => {
    const timer = setTimeout(() => setDebouncedSearch(searchText), 350)
    return () => clearTimeout(timer)
  }, [searchText])

  const { data, fetchNextPage, hasNextPage, isFetchingNextPage, isLoading, error } = useInfiniteQuery({
    queryKey: ['public-jobs', debouncedSearch, filters],
    queryFn: async ({ pageParam }) => {
      const params = new URLSearchParams()
      if (pageParam) params.set('cursor', pageParam as string)
      if (debouncedSearch) params.set('q', debouncedSearch)
      if (filters.role_type) params.set('role_type', filters.role_type)
      if (filters.seniority) params.set('seniority', filters.seniority)
      if (filters.remote) params.set('remote', 'true')
      const res = await fetch(`${API_BASE}/api/v1/public/jobs?${params}`)
      if (!res.ok) throw new Error(`Jobs fetch failed: ${res.status}`)
      return res.json() as Promise<PublicJobsResponse>
    },
    initialPageParam: null as string | null,
    getNextPageParam: (lastPage) => lastPage.next_cursor ?? undefined,
  })

  const jobs: PublicJobItem[] = data?.pages.flatMap(page => page.items) ?? []

  // IntersectionObserver for load-more (mirrors BD leads page pattern)
  useEffect(() => {
    const el = loadMoreRef.current
    if (!el) return
    const observer = new IntersectionObserver(
      (entries) => {
        if (entries[0].isIntersecting && hasNextPage && !isFetchingNextPage) fetchNextPage()
      },
      { threshold: 0.1 },
    )
    observer.observe(el)
    return () => observer.disconnect()
  }, [hasNextPage, isFetchingNextPage, fetchNextPage])

  if (isLoading) {
    return (
      <div className="px-6 py-6 space-y-6">
        <FilterBar onSearchChange={setSearchText} onFiltersChange={setFilters} />
        <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
          {Array.from({ length: 6 }).map((_, i) => (
            <Skeleton key={i} className="h-[160px] rounded-xl" />
          ))}
        </div>
      </div>
    )
  }

  if (error) {
    return (
      <div className="px-6 py-6">
        <FilterBar onSearchChange={setSearchText} onFiltersChange={setFilters} />
        <p className="mt-6 text-sm text-destructive">
          Couldn&apos;t load jobs. Check your connection or refresh.
        </p>
      </div>
    )
  }

  return (
    <div className="px-6 py-6 space-y-4">
      <FilterBar onSearchChange={setSearchText} onFiltersChange={setFilters} />

      {jobs.length === 0 ? (
        <div className="py-12 text-center">
          <p className="text-base text-foreground">No roles match your filters</p>
          <p className="mt-2 text-sm text-muted-foreground">
            Try adjusting your filters or search terms to see more opportunities.
          </p>
        </div>
      ) : (
        <div className="relative">
          <div className="grid grid-cols-1 gap-4 md:grid-cols-2 xl:grid-cols-3">
            {jobs.map((job, idx) => {
              const isLocked = !isLoggedIn && idx >= VISIBLE_LIMIT
              return (
                <JobCard
                  key={job.id}
                  job={job}
                  isLoggedIn={isLoggedIn}
                  isLocked={isLocked}
                />
              )
            })}
          </div>
          {!isLoggedIn && jobs.length > VISIBLE_LIMIT && <LockedJobsOverlay />}
        </div>
      )}

      <div ref={loadMoreRef} className="h-4" />
      {hasNextPage && (
        <div className="flex justify-center pt-2">
          <Button
            variant="outline"
            onClick={() => fetchNextPage()}
            disabled={isFetchingNextPage}
          >
            {isFetchingNextPage ? 'Loading…' : 'Load more jobs'}
          </Button>
        </div>
      )}
    </div>
  )
}
```
