function computeMockScore(body: Record<string, unknown>, base: typeof MOCK_PROFILE): number {
  const fields = [base.first_name, base.last_name, base.headline, base.location,
    (base.skills?.length ?? 0) > 0, base.remote_preference, base.seniority,
    base.github_url, base.linkedin_url, base.resume_url]
  const updated = { ...base, ...body }
  const updatedFields = [updated.first_name, updated.last_name, updated.headline, updated.location,
    ((updated.skills as string[] | undefined)?.length ?? 0) > 0, updated.remote_preference,
    updated.seniority, updated.github_url, updated.linkedin_url, updated.resume_url]
  return Math.min(100, updatedFields.filter(Boolean).length * 10)
}
