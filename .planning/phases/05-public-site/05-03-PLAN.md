```typescript
'use client'
import { useEffect } from 'react'
import { useRouter, useSearchParams } from 'next/navigation'
import { useForm } from 'react-hook-form'
import { zodResolver } from '@hookform/resolvers/zod'
import { z } from 'zod'
import { useMutation } from '@tanstack/react-query'
import { toast } from 'sonner'
import Link from 'next/link'
import { Button } from '@brynzit/ui/components/ui/button'
import { Input } from '@brynzit/ui/components/ui/input'
import { Label } from '@brynzit/ui/components/ui/label'
import { Card, CardContent, CardHeader, CardTitle, CardDescription } from '@brynzit/ui/components/ui/card'
import { useCandidateAuth } from '../_components/candidate-auth-context'
import type { CandidateAuthResponse } from '../_lib/candidate-types'

// T-05-01: validate redirect param — only allow relative paths
function safeRedirect(param: string | null): string {
  if (!param) return '/jobs'
  if (param.startsWith('/') && !param.startsWith('//')) return param
  return '/jobs'
}

const signupSchema = z.object({
  first_name: z.string().optional(),
  last_name: z.string().optional(),
  email: z.string().email('Valid email required'),
  password: z.string().min(8, 'Password must be at least 8 characters'),
})
type SignupValues = z.infer<typeof signupSchema>

const API_BASE = process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:8000'

export default function SignupPage() {
  const router = useRouter()
  const searchParams = useSearchParams()
  const { isLoggedIn, signIn } = useCandidateAuth()

  // Redirect if already logged in
  useEffect(() => {
    if (isLoggedIn) router.replace('/jobs')
  }, [isLoggedIn, router])

  const form = useForm<SignupValues>({ resolver: zodResolver(signupSchema) })

  const mutation = useMutation({
    mutationFn: async (values: SignupValues) => {
      const res = await fetch(`${API_BASE}/api/v1/candidate/auth/signup`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(values),
      })
      if (res.status === 409) throw Object.assign(new Error('conflict'), { status: 409 })
      if (!res.ok) throw new Error(`Signup failed: ${res.status}`)
      return res.json() as Promise<CandidateAuthResponse>
    },
    onSuccess: (data) => {
      signIn(data.access_token, { id: data.id, email: data.email })
      const redirect = safeRedirect(searchParams.get('redirect'))
      // After signup, always go to onboarding first; preserve redirect for after onboarding
      router.push(`/onboarding/1${redirect !== '/jobs' ? `?redirect=${encodeURIComponent(redirect)}` : ''}`)
    },
    onError: (err) => {
      if ((err as { status?: number }).status === 409) {
        toast.error('That email is already registered. Sign in instead.')
      } else {
        toast.error('Signup failed — try again.')
      }
    },
  })

  if (isLoggedIn) return null

  return (
    <main className="flex min-h-[calc(100vh-48px)] items-center justify-center px-4">
      <Card className="w-full max-w-[400px]">
        <CardHeader className="pb-4">
          <CardTitle className="text-2xl font-normal">Create your account</CardTitle>
          <CardDescription>Join the Brynzit talent pool</CardDescription>
        </CardHeader>
        <CardContent>
          <form onSubmit={form.handleSubmit((v) => mutation.mutate(v))} className="space-y-4">
            <div className="grid grid-cols-2 gap-3">
              <div>
                <Label htmlFor="first_name">First name</Label>
                <Input id="first_name" {...form.register('first_name')} />
              </div>
              <div>
                <Label htmlFor="last_name">Last name</Label>
                <Input id="last_name" {...form.register('last_name')} />
              </div>
            </div>
            <div>
              <Label htmlFor="email">Email</Label>
              <Input id="email" type="email" {...form.register('email')} />
              {form.formState.errors.email && (
                <p className="mt-1 text-xs text-destructive">{form.formState.errors.email.message}</p>
              )}
            </div>
            <div>
              <Label htmlFor="password">Password</Label>
              <Input id="password" type="password" {...form.register('password')} />
              {form.formState.errors.password && (
                <p className="mt-1 text-xs text-destructive">{form.formState.errors.password.message}</p>
              )}
            </div>
            <Button type="submit" className="w-full" disabled={mutation.isPending}>
              {mutation.isPending ? 'Creating account…' : 'Create account'}
            </Button>
            <p className="text-center text-sm text-muted-foreground">
              Already have an account?{' '}
              <Link href="/login" className="text-foreground hover:underline">Sign in</Link>
            </p>
          </form>
        </CardContent>
      </Card>
    </main>
  )
}
```

```typescript
import { useRouter, useSearchParams } from 'next/navigation'

import { zodResolver } from '@hookform/resolvers/zod'
import { z } from 'zod'
import { useMutation } from '@tanstack/react-query'
import { toast } from 'sonner'
import Link from 'next/link'

}

  email: z.string().email('Valid email required'),
```

```typescript
import { cn } from '@brynzit/ui/lib/utils'

interface StepIndicatorProps {
  currentStep: number
  totalSteps: number
}

export function StepIndicator({ currentStep, totalSteps }: StepIndicatorProps) {
  return (
    <div className="flex flex-col gap-2">
      <p className="text-sm text-muted-foreground">Step {currentStep} of {totalSteps}</p>
      <div className="flex gap-2">
        {Array.from({ length: totalSteps }).map((_, i) => (
          <div
            key={i}
            aria-current={i + 1 === currentStep ? 'step' : undefined}
            className={cn(
              'h-1 flex-1 rounded-full transition-colors',
              i + 1 < currentStep
                ? 'bg-primary/60'
                : i + 1 === currentStep
                ? 'bg-primary'
                : 'bg-muted',
            )}
          />
        ))}
      </div>
    </div>
  )
}
```

Pattern: react-dropzone + native fetch POST to /candidate/profile/resume + useAsyncJob polling. Mirror supply resume-form.tsx exactly, adapted for candidate endpoint:
```typescript
'use client'
import { useEffect, useState } from 'react'
import { useDropzone } from 'react-dropzone'
import { toast } from 'sonner'
import { Button } from '@brynzit/ui/components/ui/button'
import { Progress } from '@brynzit/ui/components/ui/progress'
import { cn } from '@brynzit/ui/lib/utils'
import { Upload } from 'lucide-react'
import { useAsyncJob } from '@brynzit/hooks'
import type { ResumeParseStatus } from '../../_lib/candidate-types'

const API_BASE = process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:8000'

interface ResumeUploadStepProps {
  onComplete: () => void
}

export function ResumeUploadStep({ onComplete }: ResumeUploadStepProps) {
  const [parseId, setParseId] = useState<string | null>(null)
  const [isUploading, setIsUploading] = useState(false)
  const [uploadError, setUploadError] = useState<string | null>(null)

  const { phase } = useAsyncJob<ResumeParseStatus>({
    jobId: parseId,
    endpoint: '/api/v1/candidate/profile/resume/status/{id}',
    statusMap: { processing: 'pending', complete: 'complete', failed: 'failed' },
    refetchInterval: 2000,
  })

  useEffect(() => {
    if (phase === 'complete') {
      toast.success('Resume parsed — your profile has been prefilled.')
      const timer = setTimeout(() => onComplete(), 800)
      return () => clearTimeout(timer)
    }
    if (phase === 'failed') {
      toast.error("We couldn't parse that file. Continue to fill in your profile manually.")
      onComplete()
    }
  }, [phase, onComplete])

  const { getRootProps, getInputProps, isDragActive } = useDropzone({
    accept: {
      'application/pdf': ['.pdf'],
      'application/msword': ['.doc'],
      'application/vnd.openxmlformats-officedocument.wordprocessingml.document': ['.docx'],
    },
    maxFiles: 1,
    onDropRejected: (rejections) => {
      setUploadError(rejections[0]?.errors[0]?.message ?? 'File type not accepted — upload PDF or DOCX.')
    },
    onDrop: async (acceptedFiles) => {
      const file = acceptedFiles[0]
      if (!file) return
      setUploadError(null)
      setIsUploading(true)
      try {
        const formData = new FormData()
        formData.append('file', file)
        // CRITICAL: native fetch — NOT apiClient.POST; browser sets multipart boundary
        const res = await fetch(`${API_BASE}/api/v1/candidate/profile/resume`, {
          method: 'POST',
          body: formData,
        })
        if (!res.ok) throw new Error(`Upload failed (${res.status})`)
        const data = await res.json() as { parse_id: string }
        setParseId(data.parse_id)
      } catch {
        toast.error("Upload failed — please try again.")
      } finally {
        setIsUploading(false)
      }
    },
  })

  const isPolling = phase === 'pending' || isUploading

  return (
    <div className="space-y-4">
      <div>
        <h2 className="text-xl font-normal text-foreground">Upload your resume</h2>
        <p className="mt-1 text-sm text-muted-foreground">
          We&apos;ll use it to prefill your profile — you can edit everything after.
        </p>
      </div>

      {isPolling ? (
        <div className="space-y-3">
```

```typescript
'use client'
import { use, useEffect } from 'react'
import { useRouter, useSearchParams } from 'next/navigation'
import { useCandidateAuth } from '../../_components/candidate-auth-context'
import { StepIndicator } from '../_components/step-indicator'
import { ResumeUploadStep } from '../_components/resume-upload-step'
import { ProfileConfirmStep } from '../_components/profile-confirm-step'
import { PreferencesStep } from '../_components/preferences-step'

function safeRedirect(param: string | null): string {
  if (!param) return '/jobs'
  if (param.startsWith('/') && !param.startsWith('//')) return param
  return '/jobs'
}

export default function OnboardingPage({ params }: { params: Promise<{ step: string }> }) {
  const { step } = use(params)
  const router = useRouter()
  const searchParams = useSearchParams()
  const { isLoggedIn } = useCandidateAuth()

  // Protected route — redirect if not logged in
  useEffect(() => {
    if (!isLoggedIn) router.replace('/login')
  }, [isLoggedIn, router])

  if (!isLoggedIn) return null

  const stepNum = parseInt(step, 10)
  const redirectParam = searchParams.get('redirect')

  function goToStep(n: number) {
    router.push(`/onboarding/${n}${redirectParam ? `?redirect=${encodeURIComponent(redirectParam)}` : ''}`)
  }

  function completeOnboarding() {
    router.push(safeRedirect(redirectParam))
  }

  return (
    <main className="flex min-h-[calc(100vh-48px)] items-start justify-center px-4 py-12">
      <div className="w-full max-w-[560px] space-y-8">
        <StepIndicator currentStep={stepNum} totalSteps={3} />

        {stepNum === 1 && (
          <ResumeUploadStep onComplete={() => goToStep(2)} />
        )}
        {stepNum === 2 && (
          <ProfileConfirmStep onComplete={() => goToStep(3)} />
        )}
        {stepNum === 3 && (
          <PreferencesStep
            onComplete={completeOnboarding}
            onSkip={completeOnboarding}
          />
        )}
      </div>
    </main>
  )
}
```
