Props: `{ onJobAdded: (job: IngestJob) => void }`.

On submit:
```typescript
const mutation = useMutation({
  mutationFn: (values: FormValues) =>
    apiClient.POST('/api/v1/supply/candidates/ingest', {
      body: { linkedin_url: values.linkedin_url } as components['schemas']['IngestLinkedInRequest'],
    }),
  onSuccess: (response) => {
    const result = response.data as components['schemas']['IngestLinkedInResponse']
    form.reset()
    toast.success(`Candidate added: ${result?.name ?? 'unknown'}.`)
    onJobAdded({
      id: crypto.randomUUID(),
      type: 'linkedin',
      label: 'LinkedIn',
      status: 'done',
      progressText: `Candidate added: ${result?.name ?? 'unknown'}`,
      candidateId: result?.candidate_id,
      timestamp: new Date(),
    })
  },
  onError: (err: unknown) => {
    // Check for 400 (Proxycurl not configured):
    const msg = (err as { status?: number })?.status === 400
      ? 'LinkedIn ingest unavailable — Proxycurl key not configured. Use CSV or resume upload instead.'
      : 'Ingest failed — try again.'
    toast.error(msg)
  },
})
```

```typescript
async function handleSubmit() {
  if (!file) return
  setIsSubmitting(true)
  setUploadError(null)
  const formData = new FormData()
  formData.append('file', file)
  try {
    const res = await fetch(
      `${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:8000'}/api/v1/supply/candidates/ingest/csv`,
      { method: 'POST', body: formData }
      // DO NOT set Content-Type header — browser sets multipart/form-data boundary
    )
    if (!res.ok) {
      const text = await res.text()
      throw new Error(`Upload failed (${res.status}). ${text.slice(0, 200)}`)
    }
    const data = await res.json() as components['schemas']['CsvIngestResponse']
    setFile(null)
    toast.success(`Imported ${data.imported} candidates.`)
    onJobAdded({
      id: crypto.randomUUID(),
      type: 'csv',
      label: 'CSV Import',
      status: 'done',
      progressText: `Imported ${data.imported} candidates, ${data.duplicates ?? 0} duplicates skipped`,
      timestamp: new Date(),
    })
  } catch (e) {
    setUploadError('CSV upload failed — check file format and try again.')
    toast.error('CSV upload failed — check file format and try again.')
  } finally {
    setIsSubmitting(false)
  }
}
```
