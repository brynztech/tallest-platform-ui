export function MSWProvider({ children }: { children: React.ReactNode }) {
  const [ready, setReady] = useState(false)

  useEffect(() => {
    async function init() {
      if (process.env.NEXT_PUBLIC_API_MOCKING === 'enabled') {
        const { worker } = await import('@brynzit/mocks/browser')
        await worker.start({ onUnhandledRequest: 'bypass' })

    init()
  }, [])

  return <>{children}</>
}

**IMPORTANT — Font loading pattern:** apps/console/app/layout.tsx uses `'use client'` for QueryClientProvider.
Next.js 16 font optimization (next/font/google) is designed for RSC layouts — loading Geist inside a
'use client' component may produce build warnings. Use the server wrapper layout pattern:
- apps/console/app/layout.tsx is a SERVER component that loads the font and exports metadata
- It renders a `<ConsoleProviders>` client component (defined inline or in a separate file) that wraps QueryClientProvider and MSWProvider
- This avoids the RSC/font warning while keeping QueryClientProvider in the tree

Server layout wrapping a client providers component:
```typescript
// apps/console/app/layout.tsx — SERVER component (no 'use client')
import type { Metadata } from 'next'
import { Geist } from 'next/font/google'
import { ConsoleProviders } from './_components/console-providers'
import './globals.css'

const geist = Geist({ subsets: ['latin'], variable: '--font-sans' })

export const metadata: Metadata = {
  title: 'Brynzit Console',
  description: 'Internal operations console',
}

export default function RootLayout({
  children,
}: {
  children: React.ReactNode
}) {
  return (
    <html lang="en" className={`dark ${geist.variable}`}>
      <body className="bg-background text-foreground font-sans antialiased">
        <ConsoleProviders>{children}</ConsoleProviders>
      </body>
    </html>
  )
}
```

```typescript
// apps/console/app/_components/console-providers.tsx — CLIENT component
'use client'

import { QueryClient, QueryClientProvider } from '@tanstack/react-query'
import { MSWProvider } from './msw-provider'

// QueryClient created outside component to avoid re-creation on renders
const queryClient = new QueryClient({
  defaultOptions: {
    queries: {
      staleTime: 30 * 1000, // 30 seconds default
      retry: 1,
    },
  },
})

export function ConsoleProviders({ children }: { children: React.ReactNode }) {
  return (
    <MSWProvider>
      <QueryClientProvider client={queryClient}>
        {children}
      </QueryClientProvider>
    </MSWProvider>
  )
}
```

This pattern: server layout loads font (RSC-compatible) → renders ConsoleProviders (client boundary) → QueryClientProvider and MSWProvider run client-side. `turbo run build --filter @brynzit/console` must produce no font-related warnings.
