### Pattern Group 1: Build Tooling — `turbo.json`

**Role:** config / batch  
**Source:** `01-RESEARCH.md` Pattern 1 (Turborepo Task Pipeline) — VERIFIED against Turborepo 2.9 docs

**Critical rule:** Use `"tasks"` key, NOT `"pipeline"`. `"pipeline"` is the Turborepo v1 legacy key; v2 silently ignores it in some contexts.

```json
{
  "$schema": "https://turbo.build/schema.json",
  "tasks": {
    "build": {
      "dependsOn": ["^build"],
      "outputs": [".next/**", "dist/**", "!.next/cache/**"]
    },
    "dev": {
      "cache": false,
      "persistent": true
    },
    "lint": {
      "dependsOn": ["^build"]
    },
    "typecheck": {
      "dependsOn": ["^build"]
    },
    "test": {
      "dependsOn": ["^build"],
      "outputs": ["coverage/**"]
    },
    "generate:api": {
      "cache": false
    },
    "clean": {
      "cache": false
    }
  }
}
```

**Rationale for `"cache": false` on `generate:api`:** The backend schema drifts; caching a stale generation is worse than a slow re-run.

**Rationale for `typecheck dependsOn: ["^build"]`:** Generated types in `packages/api-client/src/types/api.d.ts` must exist before any consuming package runs `tsc`.

### Pattern Group 3: Next.js 16 App Configuration — `apps/*/next.config.ts`

**Role:** config / batch  
**Source:** `01-RESEARCH.md` Pattern 3 — VERIFIED against Next.js 16 upgrade guide

**Critical rule:** `transpilePackages` must list ALL `@brynzit/*` packages. Without this, Next.js cannot transpile JSX/TSX in workspace packages — the build fails with cryptic module errors. `async` APIs (`headers()`, `cookies()`, `params`, `searchParams`) must all be `await`-ed; synchronous access was removed (not just deprecated) in Next.js 16.

```typescript
    '@brynzit/ui',
    '@brynzit/api-client',
    '@brynzit/mocks',
    '@brynzit/hooks',

  experimental: {
    dynamicIO: true, // enables 'use cache' directive — verify key name against Next.js 16.2 docs

export default nextConfig
```

**Assumption A1 flag:** The `dynamicIO: true` key is ASSUMED for `use cache` support. `use cache` is not used in Phase 1, so if the key is wrong, Phase 1 will not be blocked — but note it for Phase 5.

### Pattern Group 7: useAsyncJob Hook — `packages/hooks/src/use-async-job.ts`

**Role:** hook / event-driven  
**Source:** `01-RESEARCH.md` Pattern 7 — VERIFIED against TanStack Query v5 docs and live backend status vocabulary

**Critical pitfall (from RESEARCH.md Pitfall 1 + 7):** The live backend lead import model uses `"complete"` (NOT `"completed"`) as the terminal status. The D-08 default map in CONTEXT.md is missing this value. The implementation below adds `"complete": 'complete'` to the default map per the research recommendation.

```typescript
// packages/hooks/src/use-async-job.ts
import { useQuery } from '@tanstack/react-query'

export type AsyncJobPhase = 'idle' | 'pending' | 'complete' | 'failed'

export type UseAsyncJobOptions<TData, TRawStatus extends string = string> = {
  jobId: string | null | undefined    // null/undefined = idle (polling disabled)
  endpoint: string                    // supports {id} interpolation or plain string where jobId is appended
  enabled?: boolean                   // additional gate on top of jobId presence
  statusMap?: Partial<Record<TRawStatus, AsyncJobPhase>>   // override/extend default mapping
  refetchInterval?: number            // default: 2000ms
}

export type AsyncJobResult<TData, TRawStatus extends string = string> = {
  phase: AsyncJobPhase
  rawStatus?: TRawStatus
  data?: TData
  error?: unknown
  isPolling: boolean
}

/**
 * Default status vocabulary from live backend (verified 2026-06-13):
 * - "processing", "running", "generating", "queued" → pending
 * - "done", "completed", "succeeded", "success" → complete
 * - "complete" → complete  (lead import terminal state — NOT "completed")
 * - "outreach_review" → complete  (outreach generation terminal output)
 * - "failed", "error" → failed
 *
 * Callers override or extend via statusMap. Example for import polling:
 *   useAsyncJob({ ..., statusMap: { complete: 'complete' } })
 * (redundant given defaults below, but explicit callers are self-documenting)
 */
const DEFAULT_STATUS_MAP: Record<string, AsyncJobPhase> = {
  queued: 'pending',
  processing: 'pending',
  running: 'pending',
  generating: 'pending',
  enrichment_scheduled: 'pending',
  done: 'complete',
  complete: 'complete',        // lead import terminal state
  completed: 'complete',
  succeeded: 'complete',
  success: 'complete',
  outreach_review: 'complete', // outreach generation terminal — ready for human review
  failed: 'failed',
  error: 'failed',
  new: 'failed',               // candidate record after failed resume parse
}

const TERMINAL_PHASES: AsyncJobPhase[] = ['complete', 'failed']

export function useAsyncJob<TData, TRawStatus extends string = string>(
  options: UseAsyncJobOptions<TData, TRawStatus>,
): AsyncJobResult<TData, TRawStatus> {
  const {
    jobId,
    endpoint,
    enabled = true,
    statusMap = {},
    refetchInterval: intervalOverride = 2000,
  } = options

  const mergedMap = { ...DEFAULT_STATUS_MAP, ...statusMap }
  const isEnabled = !!jobId && enabled

  // Build poll URL: replace {id} placeholder or append jobId as path segment
  const url = endpoint.includes('{id}')
    ? endpoint.replace('{id}', jobId ?? '')
    : jobId
    ? `${endpoint}/${jobId}`
    : endpoint

  const query = useQuery<TData>({
    queryKey: ['async-job', url, jobId],
    queryFn: async () => {
      const res = await fetch(url)
      if (!res.ok) throw new Error(`Poll failed: ${res.status}`)
      return res.json() as Promise<TData>
    },
    enabled: isEnabled,
    // refetchInterval as function: TanStack Query v5 signature — receives query object
    refetchInterval: (q) => {
      const data = q.state.data as Record<string, unknown> | undefined
      const rawStatus = data?.status as string | undefined
      if (!rawStatus) return intervalOverride
      const phase = mergedMap[rawStatus] ?? 'pending'
      return TERMINAL_PHASES.includes(phase) ? false : intervalOverride
    },
    refetchIntervalInBackground: false,
  })

  if (!isEnabled) {
    return { phase: 'idle', isPolling: false }
  }

  const data = query.data as Record<string, unknown> | undefined
  const rawStatus = data?.status as TRawStatus | undefined
  const phase: AsyncJobPhase = rawStatus
    ? ((mergedMap[rawStatus] as AsyncJobPhase) ?? 'pending')
    : query.isError
    ? 'failed'
    : 'pending'

  return {
    phase,
    rawStatus,
    data: query.data,
    error: query.error ?? undefined,
    isPolling: query.isFetching && !TERMINAL_PHASES.includes(phase),
  }
}
```

**Key endpoint-to-poll-URL mapping (documented for callers, from RESEARCH.md Pitfall 2):**
- BD import status: `endpoint: '/api/v1/bd/imports/{id}/status'` — dedicated status endpoint
- Outreach generation: `endpoint: '/api/v1/bd/outreach/{id}'` — poll the sequence resource; `statusMap: { outreach_review: 'complete' }` (already in defaults)
- Requisition parse-jd: `endpoint: '/api/v1/ops/requisitions/{id}'` — poll the requisition
- Enrich: `endpoint: '/api/v1/supply/candidates/{id}'` — poll the candidate profile
- Resume upload: no poll endpoint — document as known gap in Phase 3/4

### Pattern Group 9: MSW Provider + instrumentation.ts — App-Level Init

**Role:** provider / event-driven  
**Source:** `01-RESEARCH.md` Pattern 5 — VERIFIED against MSW v2 mswjs.io docs

**Critical rule:** MSW init MUST be guarded by `process.env.NEXT_PUBLIC_API_MOCKING === 'enabled'`. Never import `@brynzit/mocks/browser` unconditionally — doing so will bundle MSW into the production build.

#### `apps/*/app/_components/msw-provider.tsx` (identical in both apps)

```typescript
'use client'

import { useEffect, useState } from 'react'

export function MSWProvider({ children }: { children: React.ReactNode }) {
  const [ready, setReady] = useState(false)

  useEffect(() => {
    async function init() {
      if (process.env.NEXT_PUBLIC_API_MOCKING === 'enabled') {
        const { worker } = await import('@brynzit/mocks/browser')
        await worker.start({ onUnhandledRequest: 'bypass' })
      }
      setReady(true)
    }
    init()
  }, [])

  if (!ready) return null
  return <>{children}</>
}
```

#### `apps/*/instrumentation.ts` (identical in both apps)

```typescript
export async function register() {
  if (
    process.env.NEXT_RUNTIME === 'nodejs' &&
    process.env.NEXT_PUBLIC_API_MOCKING === 'enabled'
  ) {
    const { server } = await import('@brynzit/mocks/node')
    server.listen({ onUnhandledRequest: 'bypass' })
  }
}
```

```typescript
import { describe, it, expect, vi } from 'vitest'
import { renderHook, waitFor } from '@testing-library/react'
import { QueryClient, QueryClientProvider } from '@tanstack/react-query'
import React from 'react'
import { useAsyncJob } from '../use-async-job'

function makeWrapper() {
  const queryClient = new QueryClient({
    defaultOptions: { queries: { retry: false } },
  })
  return function Wrapper({ children }: { children: React.ReactNode }) {
    return React.createElement(QueryClientProvider, { client: queryClient }, children)
  }
}

describe('useAsyncJob', () => {
  it('returns phase=idle when jobId is null', () => {
    const { result } = renderHook(
      () => useAsyncJob({ jobId: null, endpoint: '/api/v1/bd/imports/{id}/status' }),
      { wrapper: makeWrapper() },
    )
    expect(result.current.phase).toBe('idle')
    expect(result.current.isPolling).toBe(false)
  })

  it('returns phase=idle when enabled=false', () => {
    const { result } = renderHook(
      () => useAsyncJob({ jobId: 'abc123', endpoint: '/api/v1/bd/imports/{id}/status', enabled: false }),
      { wrapper: makeWrapper() },
    )
    expect(result.current.phase).toBe('idle')
  })

  it('maps backend "complete" status to phase=complete', async () => {
    global.fetch = vi.fn().mockResolvedValue({
      ok: true,
      json: async () => ({ status: 'complete', import_id: 'abc123' }),
    })

    const { result } = renderHook(
      () => useAsyncJob({ jobId: 'abc123', endpoint: '/api/v1/bd/imports/{id}/status' }),
      { wrapper: makeWrapper() },
    )

    await waitFor(() => expect(result.current.phase).toBe('complete'))
    expect(result.current.isPolling).toBe(false)
  })

  it('maps backend "processing" status to phase=pending', async () => {
    global.fetch = vi.fn().mockResolvedValue({
      ok: true,
      json: async () => ({ status: 'processing', import_id: 'abc123' }),
    })

    const { result } = renderHook(
      () => useAsyncJob({ jobId: 'abc123', endpoint: '/api/v1/bd/imports/{id}/status' }),
      { wrapper: makeWrapper() },
    )

    await waitFor(() => expect(result.current.phase).toBe('pending'))
    expect(result.current.isPolling).toBe(true)
  })

  it('custom statusMap overrides default mapping', async () => {
    global.fetch = vi.fn().mockResolvedValue({
      ok: true,
      json: async () => ({ status: 'outreach_review' }),
    })

    const { result } = renderHook(
      () => useAsyncJob({
        jobId: 'seq123',
        endpoint: '/api/v1/bd/outreach/{id}',
        statusMap: { outreach_review: 'complete' },
      }),
      { wrapper: makeWrapper() },
    )

    await waitFor(() => expect(result.current.phase).toBe('complete'))
  })
})
```
