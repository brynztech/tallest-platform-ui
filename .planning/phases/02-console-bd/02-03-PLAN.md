```typescript
'use client'

import { useState } from 'react'
import { useDropzone } from 'react-dropzone'
import { useAsyncJob } from '@brynzit/hooks'
import type { components } from '@brynzit/api-client'
import { Badge } from '@brynzit/ui/components/ui/badge'
import { Button } from '@brynzit/ui/components/ui/button'
import { cn } from '@brynzit/ui/lib/utils'
import { UploadCloud } from 'lucide-react'

type ImportStatusResponse = components['schemas']['ImportStatusResponse']

export function ImportDropzone() {
  const [importId, setImportId] = useState<string | null>(null)
  const [uploadError, setUploadError] = useState<string | null>(null)
  const [fileName, setFileName] = useState<string | null>(null)

  const { phase, data: statusData } = useAsyncJob<ImportStatusResponse>({
    jobId: importId,
    endpoint: '/api/v1/bd/imports/{id}/status',
  })

  const { getRootProps, getInputProps, isDragActive } = useDropzone({
    accept: { 'text/csv': ['.csv'] },
    maxFiles: 1,
    onDropRejected: (rejections) => {
      setUploadError(rejections[0]?.errors[0]?.message ?? 'File type not accepted — upload a CSV file.')
    },
    onDrop: async (acceptedFiles) => {
      const file = acceptedFiles[0]
      if (!file) return

      setUploadError(null)
      setFileName(file.name)

      const formData = new FormData()
      formData.append('file', file)

      try {
        // IMPORTANT: Use native fetch, NOT apiClient.POST (RESEARCH.md Open Question 4)
        const res = await fetch(
          `${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:8000'}/api/v1/bd/imports`,
          { method: 'POST', body: formData },
          // No Content-Type header — browser sets multipart/form-data boundary automatically
        )
        if (!res.ok) {
          const text = await res.text()
          setUploadError(`Upload failed (${res.status}). ${text.slice(0, 200)}`)
          return
        }
        const data = (await res.json()) as { import_id: string }
        setImportId(data.import_id)
      } catch (err) {
        setUploadError('Upload failed — check your connection.')
      }
    },
  })

  function handleReset() {
    setImportId(null)
    setFileName(null)
    setUploadError(null)
  }

  // ---- Status card (after upload) ----
  if (importId) {
    return (
      <div className="rounded-xl border border-border bg-card p-6">
        <div className="flex items-start justify-between gap-4">
          <div className="flex flex-col gap-2">
            <p className="text-sm font-medium">{fileName}</p>
            <div className="flex items-center gap-2">
              {phase === 'pending' && (
                <>
                  <Badge variant="secondary" className="flex items-center gap-1.5 text-xs">
                    <svg className="h-3 w-3 animate-spin" viewBox="0 0 24 24" fill="none">
                      <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4" />
                      <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4z" />
                    </svg>
                    Processing
                  </Badge>
                  <span className="text-sm text-muted-foreground">
                    Processing {statusData?.total ?? '…'} leads…
                  </span>
                </>
              )}
              {phase === 'complete' && (
                <>
                  <Badge
                    className="text-xs text-white"
                    style={{ backgroundColor: 'oklch(0.45 0.15 145)' }}
                  >
                    Done
                  </Badge>
                  <span className="text-sm text-muted-foreground">
                    Imported {statusData?.review_queue ?? 0} leads
                    {(statusData?.suppressed ?? 0) > 0 && `, ${statusData.suppressed} suppressed`}
                    {(statusData?.failed ?? 0) > 0 && `, ${statusData.failed} errors`}
                  </span>
                </>
              )}
              {(phase === 'failed' || phase === 'error') && (
                <>
                  <Badge variant="destructive" className="text-xs">Failed</Badge>
                  <span className="text-sm text-destructive">
                    Import failed — check the file format or contact support.
                  </span>
                </>
              )}
              {phase === 'idle' && (
                <Badge variant="outline" className="text-xs">Queued</Badge>
              )}
            </div>
          </div>
          <Button size="sm" variant="ghost" onClick={handleReset}>
            New import
          </Button>
        </div>
      </div>
    )
  }

  // ---- Dropzone ----
  return (
    <div className="flex flex-col gap-4">
      <div
        {...getRootProps()}
        className={cn(
          'flex cursor-pointer flex-col items-center justify-center rounded-xl border-2 border-dashed p-12 text-center transition-colors',
          isDragActive
            ? 'border-primary bg-primary/5 text-primary'
            : 'border-border bg-card text-muted-foreground hover:border-primary/50 hover:text-foreground',
        )}
      >
        <input {...getInputProps()} />
        <UploadCloud className="mb-3 h-8 w-8" />
        {isDragActive ? (
          <p className="text-sm font-medium">Drop CSV here</p>
        ) : (
          <>
            <p className="text-sm font-medium">Drag CSV here, or click to select</p>
            <p className="mt-1 text-xs">CSV files only, one at a time</p>
          </>
        )}
      </div>
      {uploadError && (
        <p className="text-sm text-destructive">{uploadError}</p>
      )}
    </div>
  )
}
```

```typescript
'use client'

import { useState } from 'react'
import { useMutation, useQueryClient } from '@tanstack/react-query'
import { apiClient } from '@brynzit/api-client'
import type { components } from '@brynzit/api-client'
import { Button } from '@brynzit/ui/components/ui/button'
import { Input } from '@brynzit/ui/components/ui/input'
import { Badge } from '@brynzit/ui/components/ui/badge'
import { Separator } from '@brynzit/ui/components/ui/separator'
import { cn } from '@brynzit/ui/lib/utils'
import { ChevronDown, ChevronRight } from 'lucide-react'
import { toast } from 'sonner'

// Use the full detail OutreachStepResponse schema (non-null original_subject/body)
type OutreachSequenceResponse = components['schemas']['OutreachSequenceResponse']
type OutreachStepResponse = OutreachSequenceResponse['steps'][number]

interface StepEditorProps {
  step: OutreachStepResponse
  sequenceId: string
  onStepChange: (stepNumber: number, subject: string, body: string) => void
}

function StepEditor({ step, sequenceId, onStepChange }: StepEditorProps) {
  const [subject, setSubject] = useState(step.subject ?? step.original_subject ?? '')
  const [body, setBody] = useState(step.body ?? step.original_body ?? '')
  const [isDirty, setIsDirty] = useState(false)

  const saveStep = useMutation({
    mutationFn: () =>
      apiClient.PATCH('/api/v1/bd/outreach/{sequence_id}' as never, {
        params: { path: { sequence_id: sequenceId } },
        body: { steps: [{ step_number: step.step_number, subject, body }] },
      } as never),
    onSuccess: () => {
      setIsDirty(false)
      onStepChange(step.step_number, subject, body)
    },
    onError: () => {
      toast.error('Failed to save step. Try again.')
    },
  })

  return (
    <div className="rounded-lg border border-border p-4">
      <div className="mb-3 flex items-center justify-between">
        <span className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">
          Step {step.step_number} — Day {(step.step_number - 1) * 3}
        </span>
        {isDirty && (
          <Button
            size="sm"
            variant="outline"
            className="h-7 text-xs"
            onClick={() => saveStep.mutate()}
            disabled={saveStep.isPending}
          >
            {saveStep.isPending ? 'Saving…' : 'Save'}
          </Button>
        )}
      </div>
      <div className="flex flex-col gap-3">
        <div>
          <label className="mb-1 block text-xs text-muted-foreground">Subject</label>
          <Input
            value={subject}
            onChange={(e) => { setSubject(e.target.value); setIsDirty(true) }}
            className="text-sm"
            placeholder="Subject line"
          />
        </div>
        <div>
          <label className="mb-1 block text-xs text-muted-foreground">Body</label>
          <textarea
            value={body}
            onChange={(e) => { setBody(e.target.value); setIsDirty(true) }}
            rows={5}
            className={cn(
              'flex w-full rounded-md border border-input bg-transparent px-3 py-2 text-sm',
              'placeholder:text-muted-foreground focus:outline-none focus:ring-1 focus:ring-ring',
              'resize-none',
            )}
            placeholder="Email body"
          />
        </div>
      </div>
    </div>
  )
}

interface SequenceEditorProps {
  sequence: OutreachSequenceResponse
  leadName?: string
  onApproved: (id: string) => void
  onRejected: (id: string) => void
}

export function SequenceEditor({ sequence, leadName, onApproved, onRejected }: SequenceEditorProps) {
  const [isExpanded, setIsExpanded] = useState(false)
  const [stepStates, setStepStates] = useState(
    sequence.steps.map((s) => ({ step_number: s.step_number, subject: s.subject ?? s.original_subject ?? '', body: s.body ?? s.original_body ?? '' }))
  )
  const queryClient = useQueryClient()

  const approveMutation = useMutation({
    mutationFn: () =>
      fetch(
        `${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:8000'}/api/v1/bd/outreach/${sequence.id}/approve`,
        { method: 'POST' },
      ).then((r) => { if (!r.ok) throw new Error('Approve failed'); return r.json() }),
    onSuccess: () => {
      toast('Sequence approved', { description: `${leadName ?? 'Sequence'} outreach approved.` })
      onApproved(sequence.id)
      queryClient.invalidateQueries({ queryKey: ['outreach-sequences'] })
    },
    onError: () => { toast.error('Failed to approve sequence.') },
  })

  const rejectMutation = useMutation({
    mutationFn: () =>
      fetch(
        `${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:8000'}/api/v1/bd/outreach/${sequence.id}/reject`,
        { method: 'POST' },
      ).then((r) => { if (!r.ok) throw new Error('Reject failed'); return r.json() }),
    onSuccess: () => {
      toast('Sequence rejected', { description: `${leadName ?? 'Sequence'} outreach rejected.` })
      onRejected(sequence.id)
      queryClient.invalidateQueries({ queryKey: ['outreach-sequences'] })
    },
    onError: () => { toast.error('Failed to reject sequence.') },
  })

  function handleStepChange(stepNumber: number, subject: string, body: string) {
    setStepStates((prev) => prev.map((s) => s.step_number === stepNumber ? { ...s, subject, body } : s))
  }

  const stepCount = sequence.steps.length

  return (
    <div className="rounded-xl border border-border bg-card overflow-hidden">
      {/* Collapsed header */}
      <button
        className="flex w-full items-center gap-3 px-5 py-4 text-left hover:bg-accent/30 transition-colors"
        onClick={() => setIsExpanded((v) => !v)}
        aria-expanded={isExpanded}
      >
        {isExpanded ? (
          <ChevronDown className="h-4 w-4 text-muted-foreground shrink-0" />
        ) : (
          <ChevronRight className="h-4 w-4 text-muted-foreground shrink-0" />
        )}
        <span className="flex-1 text-sm font-semibold">
          {leadName ?? `Sequence ${sequence.id.slice(0, 8)}`}
        </span>
        <span className="text-xs text-muted-foreground">{stepCount}-step sequence</span>
        <Badge variant="secondary" className="text-xs capitalize ml-2">
          {sequence.status ?? 'pending_review'}
        </Badge>
        <span className="text-xs text-muted-foreground ml-2">
          {new Date(sequence.created_at).toLocaleDateString()}
        </span>
      </button>

      {/* Expanded content */}
      {isExpanded && (
        <div className="border-t border-border">
          <div className="flex flex-col gap-4 p-5">
            {sequence.steps.map((step) => (
              <StepEditor
                key={step.step_number}
                step={step}
                sequenceId={sequence.id}
                onStepChange={handleStepChange}
              />
            ))}
          </div>
          <Separator />
          <div className="flex items-center gap-3 px-5 py-4">
            <Button
              onClick={() => approveMutation.mutate()}
              disabled={approveMutation.isPending || rejectMutation.isPending}
            >
              Approve Sequence
            </Button>
            <Button
              variant="outline"
              className="hover:bg-destructive/10 hover:text-destructive hover:border-destructive"
              onClick={() => rejectMutation.mutate()}
              disabled={approveMutation.isPending || rejectMutation.isPending}
            >
              Reject Sequence
            </Button>
          </div>
        </div>
      )}
    </div>
  )
}
```

```typescript
'use client'

import { useState } from 'react'
import { useQuery } from '@tanstack/react-query'
import type { components } from '@brynzit/api-client'
import { SequenceEditor } from './_components/sequence-editor'

type OutreachSequenceResponse = components['schemas']['OutreachSequenceResponse']

export default function OutreachPage() {
  const [removedIds, setRemovedIds] = useState<Set<string>>(new Set())

  const { data, isLoading, error } = useQuery({
    queryKey: ['outreach-sequences'],
    queryFn: async () => {
      // MSW-only: no live /api/v1/bd/outreach list endpoint (RESEARCH.md Open Question 3)
      const res = await fetch(
        `${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:8000'}/api/v1/bd/outreach`,
      )
      if (!res.ok) throw new Error('Failed to fetch outreach sequences')
      return res.json() as Promise<{ sequences: OutreachSequenceResponse[] }>
    },
  })

  if (isLoading) {
    return (
      <div className="p-6">
        <p className="text-muted-foreground text-sm">Loading…</p>
      </div>
    )
  }

  if (error) {
    return (
      <div className="p-6">
        <p className="text-destructive text-sm">Couldn't load outreach sequences.</p>
      </div>
    )
  }

  const sequences = (data?.sequences ?? []).filter((s) => !removedIds.has(s.id))

  function handleApproved(id: string) {
    setRemovedIds((prev) => new Set([...prev, id]))
  }

  function handleRejected(id: string) {
    setRemovedIds((prev) => new Set([...prev, id]))
  }

  return (
    <div className="flex flex-col gap-6 p-6">
      <div className="flex items-center justify-between">
        <h1 className="text-xl font-semibold">Outreach Review</h1>
        <span className="text-sm text-muted-foreground">{sequences.length} pending</span>
      </div>

      {sequences.length === 0 ? (
        <div className="flex flex-col items-center justify-center py-24 text-center">
          <h2 className="text-xl font-semibold">All reviewed</h2>
          <p className="mt-2 text-sm text-muted-foreground">
            No outreach sequences waiting for review.
          </p>
        </div>
      ) : (
        <div className="flex flex-col gap-3">
          {sequences.map((sequence) => (
            <SequenceEditor
              key={sequence.id}
              sequence={sequence}
              leadName={undefined} // No lead name in list — sequence.lead_id could be used for lookup
              onApproved={handleApproved}
              onRejected={handleRejected}
            />
          ))}
        </div>
      )}
    </div>
  )
}
```
