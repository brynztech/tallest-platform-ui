import { CSS } from '@dnd-kit/utilities'
import type { components } from '@brynzit/api-client'
import { Badge } from '@brynzit/ui/components/ui/badge'
import { cn } from '@brynzit/ui/lib/utils'

// IMPORTANT: Import from '@dnd-kit/sortable' — NOT '@dnd-kit/react/sortable'
// See RESEARCH.md Pitfall 5 — @dnd-kit/react is the 0.5.x pre-release rewrite

type KanbanLeadItem = { id: string; company_name: string | null; icp_score: number | null; canonical_domain: string | null }
type KanbanColumn = { status: string; count: number; leads: KanbanLeadItem[] }

const PIPELINE_STATUS_LABELS: Record<string, string> = {
  discovered: 'Discovered',
  icp_scored: 'ICP Scored',
  review_queue: 'In Review',
  outreach_approved: 'Outreach Approved',
  outreach_sent: 'Outreach Sent',
  interested: 'Replied',
  converted: 'Converted',
}

function icpBadgeClass(score: number | null) {
  if (score === null) return 'bg-muted text-muted-foreground'
  if (score >= 80) return 'bg-primary text-primary-foreground'
  if (score >= 60) return 'bg-[oklch(0.769_0.188_85.87)] text-black'
  return 'bg-muted text-muted-foreground'
}

interface KanbanCardProps {
  lead: KanbanLeadItem
}

function KanbanCard({ lead }: KanbanCardProps) {
  const { attributes, listeners, setNodeRef, transform, transition } = useSortable({ id: lead.id })

  return (
    <div
      ref={setNodeRef}
      style={{ transform: CSS.Transform.toString(transform), transition }}
      {...attributes}
      {...listeners}
      className="flex h-14 cursor-grab items-center gap-2 rounded-lg border border-border bg-card px-3 active:cursor-grabbing"
    >
      <Badge className={cn('shrink-0 text-xs px-2 py-0.5', icpBadgeClass(lead.icp_score))}>
        {lead.icp_score ?? '—'}
      </Badge>
      <span className="flex-1 truncate text-sm font-semibold">
        {lead.company_name ?? 'Unknown'}
      </span>
    </div>
  )
}

interface KanbanColumnProps {
  column: KanbanColumn
  leads: KanbanLeadItem[]
}

function KanbanColumnComponent({ column, leads }: KanbanColumnProps) {
  const leadIds = leads.map((l) => l.id)

  return (
    <div className="flex min-w-[260px] flex-col gap-2">
      <div className="flex items-center gap-2 py-1">
        <span className="text-sm font-semibold">
          {PIPELINE_STATUS_LABELS[column.status] ?? column.status}
        </span>
        <Badge variant="secondary" className="text-xs text-muted-foreground">
          {column.count}
        </Badge>
      </div>
      <SortableContext items={leadIds} strategy={verticalListSortingStrategy}>
        <div className="flex flex-col gap-1.5 min-h-[60px]">
          {leads.map((lead) => (
            <KanbanCard key={lead.id} lead={lead} />
          ))}
        </div>
      </SortableContext>
    </div>
  )
}

interface KanbanBoardProps {
  columns: KanbanColumn[]
}

export function KanbanBoard({ columns: initialColumns }: KanbanBoardProps) {
  // Local state for visual-only reordering (no API calls — visual drag only)
  const [columns, setColumns] = useState<Record<string, KanbanLeadItem[]>>(
    Object.fromEntries(initialColumns.map((col) => [col.status, col.leads])),
  )

  const sensors = useSensors(
    useSensor(PointerSensor),
    useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates }),
  )

  function handleDragEnd(event: DragEndEvent) {
    const { active, over } = event
    if (!over || active.id === over.id) return

    // Find which column the drag occurred in
    const sourceColumn = Object.entries(columns).find(([, leads]) =>
      leads.some((l) => l.id === active.id),
    )
    if (!sourceColumn) return

    const [colStatus, colLeads] = sourceColumn
    const oldIndex = colLeads.findIndex((l) => l.id === active.id)
    const newIndex = colLeads.findIndex((l) => l.id === over.id)

    if (oldIndex === -1 || newIndex === -1) return

    // Visual-only reorder — no API persistence
    setColumns((prev) => ({
      ...prev,
      [colStatus]: arrayMove(colLeads, oldIndex, newIndex),
    }))
  }

  return (
    <DndContext sensors={sensors} collisionDetection={closestCenter} onDragEnd={handleDragEnd}>
      <div className="flex gap-4 overflow-x-auto pb-4">
        {initialColumns.map((col) => (
          <KanbanColumnComponent
            key={col.status}
            column={col}
            leads={columns[col.status] ?? col.leads}
          />
        ))}
      </div>
    </DndContext>
  )
}

import { useQuery } from '@tanstack/react-query'
import { apiClient } from '@brynzit/api-client'

import { KanbanBoard } from './_components/kanban-board'

export default function PipelinePage() {
  const { data, isLoading, error } = useQuery({
    queryKey: ['pipeline'],
    queryFn: async () => {
      const { data, error } = await apiClient.GET('/api/v1/bd/pipeline')
      if (error) throw error
      return data
    },
    refetchInterval: 15_000, // Poll every 15 seconds for live counts
  })

  if (isLoading) {
    return (
      <div className="p-6">
        <p className="text-muted-foreground text-sm">Loading…</p>
      </div>
    )
  }

  if (error) {
    return (
      <div className="p-6">
        <p className="text-destructive text-sm">Couldn't load pipeline data.</p>
      </div>
    )
  }

  return (
    <div className="flex flex-col gap-6 p-6">
      <div className="flex items-center justify-between">
        <h1 className="text-xl font-semibold">Pipeline</h1>
      </div>
      {data?.columns && <KanbanBoard columns={data.columns as any} />}
    </div>
  )

            </Button>
          </>
        )}
        {task.classification === 'ooo' && (
          <Button size="sm" variant="outline" onClick={() => handleSnooze(0, 2)}>
            Snooze 2 weeks
          </Button>
        )}
        {task.classification === 'not_now' && (
          <Button size="sm" variant="outline" onClick={() => handleSnooze(1)}>
            Snooze 1 month
          </Button>
        )}
        {task.classification === 'unsubscribe' && (
          <Button size="sm" variant="destructive" onClick={handleUnsubscribe}>
            Unsubscribe
          </Button>
        )}
      </div>
    </div>
  )
}

export default function RepliesPage() {
  const [localHandledIds, setLocalHandledIds] = useState(new Set<string>())

  const { data, isLoading, error } = useQuery({
    queryKey: ['bd-tasks'],
    queryFn: async () => {
      const res = await fetch(
        `${process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:8000'}/api/v1/bd/tasks`,
      )
      if (!res.ok) throw new Error('Failed to fetch reply tasks')
      return res.json() as Promise<{ tasks: BDReplyTask[] }>
    },
  })

  if (isLoading) {
    return (
      <div className="p-6">
        <p className="text-muted-foreground text-sm">Loading…</p>
      </div>
    )
  }

  if (error) {
    return (
      <div className="p-6">
        <p className="text-destructive text-sm">Couldn't load reply tasks.</p>
      </div>
    )
  }

  const allTasks = data?.tasks ?? []
  const tasks = allTasks
    .filter((t) => !localHandledIds.has(t.id))
    .sort((a, b) => {
      const priority: Record<string, number> = { interested: 0, ooo: 1, not_now: 2, unsubscribe: 3 }
      return (priority[a.classification] ?? 9) - (priority[b.classification] ?? 9)
    })

  const interestedCount = tasks.filter((t) => t.classification === 'interested').length

  function handleTaskDone(id: string) {
    setLocalHandledIds((prev) => new Set([...prev, id]))
  }

  return (
    <div className="flex flex-col gap-6 p-6">
      <div className="flex items-center gap-3">
        <h1 className="text-xl font-semibold">Replies</h1>
        {interestedCount > 0 && (
          <Badge className="border border-primary/30 bg-primary/20 text-primary text-xs">
            {interestedCount} interested
          </Badge>
        )}
        <span className="ml-auto text-sm text-muted-foreground">{tasks.length} pending</span>
      </div>

      {tasks.length === 0 ? (
        <div className="flex flex-col items-center justify-center py-24 text-center">
          <h2 className="text-xl font-semibold">All clear</h2>
          <p className="mt-2 text-sm text-muted-foreground">No reply tasks waiting for action.</p>
        </div>
      ) : (
        <div className="flex flex-col gap-3">
          {tasks.map((task) => (
            <ReplyCard key={task.id} task={task} onHandled={handleTaskDone} />
          ))}
        </div>
      )}
    </div>
  )
}
