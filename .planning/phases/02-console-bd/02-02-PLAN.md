```typescript
'use client'

import { useQuery } from '@tanstack/react-query'
import { apiClient } from '@brynzit/api-client'
import { Popover, PopoverContent, PopoverTrigger } from '@brynzit/ui/components/ui/popover'
import { Button } from '@brynzit/ui/components/ui/button'
import { Separator } from '@brynzit/ui/components/ui/separator'

// Display labels for SCREAMING_SNAKE_CASE enum values (from UI-SPEC copywriting contract)
const REASON_LABELS: Record<string, string> = {
  NOT_ICP_FIT: 'Not a fit',
  WRONG_SENIORITY: 'Wrong seniority',
  NO_OPEN_ROLES: 'No open roles',
  COMPETITOR: 'Competitor',
  BAD_TIMING: 'Bad timing',
  ALREADY_A_CLIENT: 'Already a client',
  OTHER: 'Other',
}

interface RejectPopoverProps {
  leadId: string
  trigger: React.ReactNode
  open: boolean
  onOpenChange: (open: boolean) => void
  onReject: (leadId: string, reason: string) => void
}

export function RejectPopover({ leadId, trigger, open, onOpenChange, onReject }: RejectPopoverProps) {
  const { data } = useQuery({
    queryKey: ['rejection-reasons'],
    queryFn: async () => {
      const { data, error } = await apiClient.GET('/api/v1/bd/rejection-reasons')
      if (error) throw error
      return data
    },
    staleTime: Infinity, // Static data — never refetch
  })

  const reasons = data?.reasons ?? []

  return (
    <Popover open={open} onOpenChange={onOpenChange}>
      <PopoverTrigger asChild>{trigger}</PopoverTrigger>
      <PopoverContent side="bottom" align="end" sideOffset={4} className="w-64 p-1">
        <div className="px-3 py-2 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
          Rejection reason
        </div>
        <Separator />
        <div className="mt-1 flex flex-col">
          {reasons.map((reason) => (
            <button
              key={reason}
              className="w-full rounded-md px-3 py-2 text-left text-sm hover:bg-accent hover:text-accent-foreground transition-colors"
              onClick={() => {
                onReject(leadId, reason)
                onOpenChange(false)
              }}
            >
              {REASON_LABELS[reason] ?? reason}
            </button>
          ))}
        </div>
      </PopoverContent>
    </Popover>
  )
}
```

```typescript
'use client'

import { useState } from 'react'
import { Popover, PopoverContent, PopoverTrigger } from '@brynzit/ui/components/ui/popover'
import { Input } from '@brynzit/ui/components/ui/input'
import { Separator } from '@brynzit/ui/components/ui/separator'

function getPresetDate(preset: 'week' | 'month' | 'quarter'): string {
  const d = new Date()
  if (preset === 'week') d.setDate(d.getDate() + 7)
  else if (preset === 'month') d.setMonth(d.getMonth() + 1)
  else {
    const q = Math.floor(d.getMonth() / 3)
    d.setMonth((q + 1) * 3, 0)
  }
  return d.toISOString()
}

interface SnoozePopoverProps {
  leadId: string
  trigger: React.ReactNode
  open: boolean
  onOpenChange: (open: boolean) => void
  onSnooze: (leadId: string, dueAt: string) => void
}

export function SnoozePopover({ leadId, trigger, open, onOpenChange, onSnooze }: SnoozePopoverProps) {
  const [showCustom, setShowCustom] = useState(false)
  const [customDate, setCustomDate] = useState('')

  function handlePreset(preset: 'week' | 'month' | 'quarter') {
    onSnooze(leadId, getPresetDate(preset))
    onOpenChange(false)
    setShowCustom(false)
  }

  function handleCustomConfirm() {
    if (!customDate) return
    onSnooze(leadId, new Date(customDate).toISOString())
    onOpenChange(false)
    setShowCustom(false)
  }

  return (
    <Popover open={open} onOpenChange={(o) => { onOpenChange(o); if (!o) setShowCustom(false) }}>
      <PopoverTrigger asChild>{trigger}</PopoverTrigger>
      <PopoverContent side="bottom" align="end" sideOffset={4} className="w-56 p-1">
        <div className="px-3 py-2 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
          Snooze until…
        </div>
        <Separator />
        <div className="mt-1 flex flex-col gap-0.5">
          {(['week', 'month', 'quarter'] as const).map((preset) => (
            <button
              key={preset}
              className="w-full rounded-md px-3 py-2 text-left text-sm hover:bg-accent hover:text-accent-foreground transition-colors"
              onClick={() => handlePreset(preset)}
            >
              {preset === 'week' ? '1 week' : preset === 'month' ? '1 month' : 'End of quarter'}
            </button>
          ))}
          <button
            className="w-full rounded-md px-3 py-2 text-left text-sm hover:bg-accent hover:text-accent-foreground transition-colors"
            onClick={() => setShowCustom(true)}
          >
            Custom date…
          </button>
          {showCustom && (
            <div className="mt-2 flex gap-2 px-2">
              <Input
                type="date"
                value={customDate}
                onChange={(e) => setCustomDate(e.target.value)}
                className="flex-1 text-sm h-8"
                aria-label="Custom snooze date"
              />
              <button
                className="rounded-md bg-primary px-2 py-1 text-xs text-primary-foreground hover:bg-primary/90"
                onClick={handleCustomConfirm}
              >
                Set
              </button>
            </div>
          )}
        </div>
      </PopoverContent>
    </Popover>
  )
}
```

```typescript
function icpBadgeClass(score: number | null) {
  if (!score) return 'bg-muted text-muted-foreground'
  if (score >= 80) return 'bg-primary text-primary-foreground'
  if (score >= 60) return 'bg-[oklch(0.769_0.188_85.87)] text-black'
  return 'bg-muted text-muted-foreground'
}

interface LeadDetailPanelProps {
  leadId: string | null
  isOpen: boolean
  onOpenChange: (open: boolean) => void
  onApprove: (id: string) => void
  onReject: (id: string, reason: string) => void
  onSnooze: (id: string, dueAt: string) => void
}

export function LeadDetailPanel({ leadId, isOpen, onOpenChange, onApprove, onReject, onSnooze }: LeadDetailPanelProps) {
  const [isRejectOpen, setIsRejectOpen] = useState(false)
  const [isSnoozeOpen, setIsSnoozeOpen] = useState(false)

  const { data, isLoading } = useQuery({
    queryKey: ['lead-detail', leadId],
    queryFn: async () => {
      const { data, error } = await apiClient.GET('/api/v1/bd/leads/{lead_id}', {
        params: { path: { lead_id: leadId! } },
      })
      if (error) throw error
      return data
    },
    enabled: !!leadId && isOpen,
    staleTime: 30 * 1000,
  })

  return (
    <Sheet open={isOpen} onOpenChange={onOpenChange}>
      <SheetContent
        side="right"
        className="w-[560px] max-w-[90vw] overflow-y-auto bg-card border-l border-border p-0"

        {/* Sticky header */}
        <div className="sticky top-0 z-10 flex items-start gap-3 border-b border-border bg-card/95 px-6 py-4 backdrop-blur">
          {isLoading ? (
            <Skeleton className="h-6 w-48" />
          ) : (
            <>
              <Badge className={cn('text-xs/600 px-2.5 py-1', icpBadgeClass(data?.icp_score ?? null))}>
                {data?.icp_score ?? '—'}
              </Badge>
              <div className="flex-1 min-w-0">
                <SheetTitle className="text-xl font-semibold leading-tight">
                  {data?.company_name ?? 'Unknown Company'}
                </SheetTitle>

                  <a
                    href={`https://${data.canonical_domain}`}
                    target="_blank"
                    rel="noopener noreferrer"
                    className="mt-0.5 flex items-center gap-1 text-sm text-primary hover:underline"
                  >
                    {data.canonical_domain}
                    <ExternalLink className="h-3 w-3" />
                  </a>
                )}
              </div>
            </>
          )}
        </div>

        {/* Action bar */}
        {leadId && (
          <div className="flex items-center gap-2 border-b border-border px-6 py-3">
            <Button size="sm" onClick={() => { onApprove(leadId); onOpenChange(false) }}>
              Approve Lead

              leadId={leadId}
              open={isRejectOpen}
              onOpenChange={setIsRejectOpen}
              onReject={(id, reason) => { onReject(id, reason); onOpenChange(false) }}

            <div className="flex flex-col gap-3">

                  <span>{data.contact_name ?? '—'}</span>
                  <span className="text-muted-foreground">{data.contact_title ?? '—'}</span>

                    <a href={`mailto:${data.contact_email}`} className="text-primary hover:underline">
```

```typescript
'use client'

import { useEffect, useRef, useState, useCallback } from 'react'
import { useInfiniteQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import type { InfiniteData } from '@tanstack/react-query'
import { apiClient } from '@brynzit/api-client'
import type { components } from '@brynzit/api-client'
import { toast } from 'sonner'
import { LeadCard } from './_components/lead-card'
import { LeadDetailPanel } from './_components/lead-detail-panel'
import { BulkToolbar } from './_components/bulk-toolbar'
import { ShortcutOverlay } from './_components/shortcut-overlay'

type LeadQueueResponse = components['schemas']['LeadQueueResponse']
type LeadQueueItem = components['schemas']['LeadQueueItem']

export default function LeadsPage() {
  const queryClient = useQueryClient()

  // ---- Infinite query (cursor pagination) ----
  const { data, fetchNextPage, hasNextPage, isFetchingNextPage, isLoading, error } = useInfiniteQuery({
    queryKey: ['leads'],
    queryFn: async ({ pageParam }) => {
      const { data, error } = await apiClient.GET('/api/v1/bd/leads', {
        params: {
          query: {
            status: 'review_queue',
            cursor: pageParam ?? undefined,
            limit: 50,
          },
        },
      })
      if (error) throw error
      return data
    },
    initialPageParam: null as string | null,
    getNextPageParam: (lastPage) => lastPage.next_cursor ?? undefined,
  })

  const leads: LeadQueueItem[] = data?.pages.flatMap((page) => page.leads) ?? []
  const totalCount = data?.pages[0]?.total ?? 0

  // ---- UI state ----
  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
  const [leavingIds, setLeavingIds] = useState<Set<string>>(new Set())
  const [processingIds, setProcessingIds] = useState<Set<string>>(new Set())
  const [focusedIndex, setFocusedIndex] = useState<number>(0)
  const [detailLeadId, setDetailLeadId] = useState<string | null>(null)
  const [isDetailOpen, setIsDetailOpen] = useState(false)
  const [isShortcutOpen, setIsShortcutOpen] = useState(false)
  const [anyPopoverOpen, setAnyPopoverOpen] = useState(false)

  // Undo timers: leadId -> timeoutRef
  const undoTimers = useRef<Map<string, ReturnType<typeof setTimeout>>>(new Map())

  // ---- Mutations ----
  const actionMutation = useMutation({
    mutationFn: (payload: components['schemas']['BulkActionRequest']) =>
      apiClient.POST('/api/v1/bd/leads/bulk-action', { body: payload }),
    onMutate: async (payload) => {
      await queryClient.cancelQueries({ queryKey: ['leads'] })
      const previousData = queryClient.getQueryData(['leads'])
      queryClient.setQueryData(['leads'], (old: InfiniteData<LeadQueueResponse> | undefined) => {
        if (!old) return old
        return {
          ...old,
          pages: old.pages.map((page) => ({
            ...page,
            leads: page.leads.filter((l) => !payload.lead_ids.includes(l.id)),
          })),
        }
      })
      return { previousData }
    },
    onError: (_err, payload, context) => {
      if (context?.previousData) {
        queryClient.setQueryData(['leads'], context.previousData)
      }
      // Remove from leavingIds to restore card
      setLeavingIds((prev) => {
        const next = new Set(prev)
        payload.lead_ids.forEach((id) => next.delete(id))
        return next
      })
      toast.error('Action failed — lead restored.')
    },
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: ['leads'] })
    },
  })

  function executeAction(payload: components['schemas']['BulkActionRequest']) {
    actionMutation.mutate(payload)
  }

  function handleAction(leadId: string, action: 'approve' | 'reject' | 'snooze', extra?: { reason?: string; dueAt?: string }) {
    // Optimistic exit
    setLeavingIds((prev) => new Set([...prev, leadId])  )

    const lead = leads.find((l) => l.id === leadId)
    const companyName = lead?.company_name ?? 'Lead'

    let toastMessage = ''
    if (action === 'approve') toastMessage = `Approved — ${companyName}`
    else if (action === 'reject') toastMessage = `Rejected — ${extra?.reason ? REASON_LABELS[extra.reason] ?? extra.reason : 'reason'}`
    else toastMessage = extra?.dueAt
      ? `Snoozed until ${new Date(extra.dueAt).toLocaleDateString()}`
      : 'Snoozed'

    const toastId = toast(toastMessage, {
      action: {
        label: 'Undo',
        onClick: () => {
          const timer = undoTimers.current.get(leadId)
          if (timer) {
            clearTimeout(timer)
            undoTimers.current.delete(leadId)
          }
          setLeavingIds((prev) => {
            const next = new Set(prev)
            next.delete(leadId)
            return next
          })
          // Restore from cache (invalidate to refetch)
          queryClient.invalidateQueries({ queryKey: ['leads'] })
          toast.dismiss(toastId)
        },
      },
      duration: 5000,
    })

    // Fire API after 300ms (allows undo before network)
    const timer = setTimeout(() => {
      undoTimers.current.delete(leadId)
      const payload: components['schemas']['BulkActionRequest'] = {
        action,
        lead_ids: [leadId],
        ...(action === 'reject' && extra?.reason ? { rejection_reason: extra.reason } : {}),
        ...(action === 'snooze' && extra?.dueAt ? { due_at: extra.dueAt } : {}),
      }
      executeAction(payload)
    }, 300)
    undoTimers.current.set(leadId, timer)
  }

  const REASON_LABELS: Record<string, string> = {
    NOT_ICP_FIT: 'Not a fit',
    WRONG_SENIORITY: 'Wrong seniority',
    NO_OPEN_ROLES: 'No open roles',
    COMPETITOR: 'Competitor',
    BAD_TIMING: 'Bad timing',
    ALREADY_A_CLIENT: 'Already a client',
    OTHER: 'Other',
  }

  // ---- Bulk actions ----
  function handleBulkStart(ids: string[]) {
    setProcessingIds(new Set(ids))
  }

  function handleBulkSettled() {
    setProcessingIds(new Set())
    setSelectedIds(new Set())
  }

  // ---- Keyboard shortcuts (CUX-03) ----
  const visibleLeads = leads.filter((l) => !leavingIds.has(l.id))

  useEffect(() => {
    function handleKeyDown(e: KeyboardEvent) {
      const activeEl = document.activeElement
      const isInputFocused =
        activeEl instanceof HTMLInputElement ||
        activeEl instanceof HTMLTextAreaElement ||
        (activeEl as HTMLElement)?.isContentEditable
      if (isInputFocused || isDetailOpen || isShortcutOpen || anyPopoverOpen) return

      switch (e.key) {
        case 'j': {
          e.preventDefault()
          setFocusedIndex((i) => Math.min(i + 1, visibleLeads.length - 1))
          break
        }
        case 'k': {
          e.preventDefault()
          setFocusedIndex((i) => Math.max(i - 1, 0))
          break
        }
        case 'a': {
          e.preventDefault()
          const lead = visibleLeads[focusedIndex]
          if (lead) handleAction(lead.id, 'approve')
          break
        }
        case 'r': {
          e.preventDefault()
          // Open reject popover on focused card — handled by card component via data attribute
          const cardEl = document.querySelector(`[data-lead-id="${visibleLeads[focusedIndex]?.id}"] [data-reject-trigger]`)
          if (cardEl instanceof HTMLElement) cardEl.click()
          break
        }
        case 's': {
          e.preventDefault()
          const snoozeEl = document.querySelector(`[data-lead-id="${visibleLeads[focusedIndex]?.id}"] [data-snooze-trigger]`)
          if (snoozeEl instanceof HTMLElement) snoozeEl.click()
          break
        }
        case '?': {
          e.preventDefault()
          setIsShortcutOpen(true)
          break
        }
        case 'Enter': {
          e.preventDefault()
          const lead = visibleLeads[focusedIndex]
          if (lead) { setDetailLeadId(lead.id); setIsDetailOpen(true) }
          break
        }
        case ' ': {
          e.preventDefault()
          const lead = visibleLeads[focusedIndex]
          if (lead) {
            setSelectedIds((prev) => {
              const next = new Set(prev)
              next.has(lead.id) ? next.delete(lead.id) : next.add(lead.id)
              return next
            })
          }
          break
        }
      }
    }
    window.addEventListener('keydown', handleKeyDown)
    return () => window.removeEventListener('keydown', handleKeyDown)
  }, [isDetailOpen, isShortcutOpen, anyPopoverOpen, focusedIndex, visibleLeads])

  // ---- Intersection observer for "load more" ----
  const loadMoreRef = useRef<HTMLDivElement>(null)
  useEffect(() => {
    const el = loadMoreRef.current
    if (!el) return
    const observer = new IntersectionObserver(
      (entries) => {
        if (entries[0].isIntersecting && hasNextPage && !isFetchingNextPage) {
          fetchNextPage()
        }
      },
      { threshold: 0.1 },
    )
    observer.observe(el)
    return () => observer.disconnect()
  }, [hasNextPage, isFetchingNextPage, fetchNextPage])

  // ---- Render ----
  if (isLoading) {
    return (
      <div className="p-6">
        <p className="text-muted-foreground text-sm">Loading…</p>
      </div>
    )
  }

  if (error) {
    return (
      <div className="p-6">
        <p className="text-destructive text-sm">Couldn't load leads. Check your connection or refresh the page.</p>
      </div>
    )
  }

  return (
    <div className="flex flex-col">
      {/* Sticky toolbar */}
      <div className="sticky top-0 z-10 flex h-14 items-center gap-4 border-b border-border bg-background/90 px-6 backdrop-blur">
        <h1 className="text-xl font-semibold">Lead Queue</h1>
        <span className="text-sm text-muted-foreground">{totalCount} leads</span>
        {selectedIds.size > 0 && (
          <div className="ml-auto">
            <BulkToolbar
              selectedIds={[...selectedIds]}
              onClearSelection={() => setSelectedIds(new Set())}
              onBulkStart={handleBulkStart}
              onBulkSettled={handleBulkSettled}
            />
          </div>
        )}
      </div>

      {/* Lead cards */}
      <div className="flex flex-col gap-2 px-4 py-4">
        {visibleLeads.length === 0 && !isLoading ? (
          <div className="flex flex-col items-center justify-center py-24 text-center">
            <h2 className="text-xl font-semibold">Queue is clear</h2>
            <p className="mt-2 max-w-sm text-sm text-muted-foreground">
              No leads waiting for review. New leads appear here when the ICP scoring agent processes them.
            </p>
          </div>
        ) : (
          visibleLeads.map((lead, idx) => (
            <LeadCard
              key={lead.id}
              lead={lead}
              isSelected={selectedIds.has(lead.id)}
              isLeaving={leavingIds.has(lead.id)}
              isProcessing={processingIds.has(lead.id)}
              isFocused={idx === focusedIndex}
              onSelect={(id, checked) => {
                setSelectedIds((prev) => {
                  const next = new Set(prev)
                  checked ? next.add(id) : next.delete(id)
                  return next
                })
              }}
              onOpenDetail={(id) => { setDetailLeadId(id); setIsDetailOpen(true) }}
              onApprove={(id) => handleAction(id, 'approve')}
              onReject={(id, reason) => handleAction(id, 'reject', { reason })}
              onSnooze={(id, dueAt) => handleAction(id, 'snooze', { dueAt })}
              onPopoverOpenChange={setAnyPopoverOpen}
            />
          ))
        )}

        {/* Load more trigger */}
        <div ref={loadMoreRef} className="h-4" />
        {isFetchingNextPage && (
          <p className="py-2 text-center text-sm text-muted-foreground">Loading more…</p>
        )}
      </div>

      {/* Side-over detail panel */}
      <LeadDetailPanel
        leadId={detailLeadId}
        isOpen={isDetailOpen}
        onOpenChange={setIsDetailOpen}
        onApprove={(id) => handleAction(id, 'approve')}
        onReject={(id, reason) => handleAction(id, 'reject', { reason })}
        onSnooze={(id, dueAt) => handleAction(id, 'snooze', { dueAt })}
      />

      {/* Shortcut overlay */}
      <ShortcutOverlay open={isShortcutOpen} onOpenChange={setIsShortcutOpen} />
    </div>
  )
}
```
