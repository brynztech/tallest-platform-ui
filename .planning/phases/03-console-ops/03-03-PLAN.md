```typescript
export function useAsyncJob<TData = unknown>({
  jobId,           // string | null — null = idle (polling disabled)
  endpoint,        // string — URL template e.g. '/api/v1/ops/requisitions/{id}'
  statusMap?,      // Record<string, AsyncJobPhase> — overrides DEFAULT_STATUS_MAP
  refetchInterval? // number — ms between polls (default 2000)
}: UseAsyncJobOptions<TData>): { phase: AsyncJobPhase; data: TData | undefined }
// AsyncJobPhase = 'idle' | 'pending' | 'complete' | 'failed'
// DEFAULT_STATUS_MAP does NOT include 'open' — always pass the statusMap override for JD parse
```

CRITICAL — statusMap for JD parse (Pitfall 1 from RESEARCH.md):
```typescript
statusMap: {
  open: 'complete',     // RequisitionResponse.status after parse completes
  filled: 'complete',   // closed requisition
  closed: 'complete',   // closed requisition
  parsing: 'pending',   // if backend uses a 'parsing' status during parse
}
```
Without this override, `phase` stays `'pending'` forever because `'open'` is not in DEFAULT_STATUS_MAP.

```typescript
const createMutation = useMutation({
  mutationFn: async (body: components['schemas']['CreateRequisitionRequest']) => {
    const { data, error } = await apiClient.POST('/api/v1/ops/requisitions', { body })
    if (error) throw error
    return data
  },
  onSuccess: async (req) => {
    setReqId(req.id)
    if (jdText.trim()) {
      const { error } = await apiClient.POST('/api/v1/ops/requisitions/{req_id}/parse-jd', {
        params: { path: { req_id: req.id } },
        body: { jd_text: jdText },

      if (!error) {
        setIsParsingJd(true)
      } else {
        // parse-jd POST failed — navigate directly

      setIsCreateOpen(false)
```

```typescript
    return <Badge variant="outline" className="text-xs text-muted-foreground border-border">{status}</Badge>
  }
  return <Badge variant="secondary" className="text-xs">{status}</Badge>
}
```

Props: `interface RequisitionFieldsProps { req: components['schemas']['RequisitionResponse'] }`

This component displays the fields card for the requisition. It also has the Edit dialog (PATCH) for updating fields. No JD parse polling here — that happens in the create flow on the list page. The fields are static displays that can be edited.

```typescript
'use client'

import { useState } from 'react'
import { useMutation, useQueryClient } from '@tanstack/react-query'
import { apiClient } from '@brynzit/api-client'
import type { components } from '@brynzit/api-client'
import { useForm } from 'react-hook-form'
import { zodResolver } from '@hookform/resolvers/zod'
import { z } from 'zod'
import { toast } from 'sonner'
import { Button } from '@brynzit/ui/components/ui/button'
import { Badge } from '@brynzit/ui/components/ui/badge'
import { Input } from '@brynzit/ui/components/ui/input'
import { Label } from '@brynzit/ui/components/ui/label'
import {
  Dialog, DialogContent, DialogHeader, DialogTitle,
} from '@brynzit/ui/components/ui/dialog'
import { Pencil } from 'lucide-react'

interface RequisitionFieldsProps {
  req: components['schemas']['RequisitionResponse']
}

const editReqSchema = z.object({
  title: z.string().min(1, 'Title is required'),
  seniority: z.string().nullable().optional(),
  skills: z.string().nullable().optional(),    // comma-separated string → split to array before POST
  compensation_range: z.string().nullable().optional(),
  target_start_date: z.string().nullable().optional(),
})
type EditReqFormValues = z.infer<typeof editReqSchema>

export function RequisitionFields({ req }: RequisitionFieldsProps) {
  const [isEditOpen, setIsEditOpen] = useState(false)
  const queryClient = useQueryClient()

  const form = useForm<EditReqFormValues>({
    resolver: zodResolver(editReqSchema),
    defaultValues: {
      title: req.title,
      seniority: req.seniority ?? '',
      skills: (req.skills ?? []).join(', '),
      compensation_range: req.compensation_range ?? '',
      target_start_date: req.target_start_date ?? '',
    },
  })

  const patchMutation = useMutation({
    mutationFn: (values: EditReqFormValues) => {
      const skillsArray = values.skills
        ? values.skills.split(',').map(s => s.trim()).filter(Boolean)
        : []
      return apiClient.PATCH('/api/v1/ops/requisitions/{req_id}', {
        params: { path: { req_id: req.id } },
        body: {
          title: values.title,
          seniority: values.seniority || null,
          skills: skillsArray.length > 0 ? skillsArray : null,
          compensation_range: values.compensation_range || null,
          target_start_date: values.target_start_date || null,
        } satisfies Partial<components['schemas']['CreateRequisitionRequest']>,
      })
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['ops-requisition', req.id] })
      toast('Requisition updated.')
      setIsEditOpen(false)
    },
    onError: () => { toast.error('Failed to update requisition.') },
  })

  const skills = req.skills ?? []
  const hasJdData = req.seniority || (req.skills && req.skills.length > 0) || req.compensation_range

  return (
    <div className="rounded-lg border border-border bg-card p-4 mb-4">
      <div className="flex items-center justify-between mb-4">
        <h2 className="text-base font-semibold">Details</h2>
        <Button variant="ghost" size="sm" onClick={() => setIsEditOpen(true)}>
          <Pencil className="size-4 mr-1" />Edit
        </Button>
      </div>

      <div className="grid grid-cols-2 gap-4">
        <div>
          <p className="text-xs text-muted-foreground mb-1">Seniority</p>
          <p className="text-sm">{req.seniority ?? '—'}</p>
        </div>
        <div>
          <p className="text-xs text-muted-foreground mb-1">Compensation Range</p>
          <p className="text-sm">{req.compensation_range ?? '—'}</p>
        </div>
        <div>
          <p className="text-xs text-muted-foreground mb-1">Target Start Date</p>
          <p className="text-sm">{req.target_start_date ?? '—'}</p>
        </div>
        <div>
          <p className="text-xs text-muted-foreground mb-1">Skills</p>
          {skills.length > 0 ? (
            <div className="flex flex-wrap gap-1">
              {skills.map((skill) => (
                <Badge key={skill} variant="secondary" className="text-xs">{skill}</Badge>
              ))}
            </div>
          ) : (
            <p className="text-sm">—</p>
          )}
        </div>
      </div>

      {!hasJdData && (
        <p className="mt-4 text-xs text-muted-foreground italic">
          No JD parsed — fields were entered manually.
        </p>
      )}

      {/* Edit Dialog */}
      <Dialog open={isEditOpen} onOpenChange={setIsEditOpen}>
        <DialogContent className="max-w-lg">
          <DialogHeader>
            <DialogTitle>Edit Requisition</DialogTitle>
          </DialogHeader>
          <form onSubmit={form.handleSubmit((v) => patchMutation.mutate(v))} className="flex flex-col gap-4">
            <div>
              <Label htmlFor="title">Title</Label>
              <Input id="title" {...form.register('title')} className="mt-1" />
            </div>
            <div>
              <Label htmlFor="seniority">Seniority</Label>
              <Input id="seniority" {...form.register('seniority')} placeholder="e.g. senior" className="mt-1" />
            </div>
            <div>
              <Label htmlFor="skills">Skills (comma-separated)</Label>
              <Input id="skills" {...form.register('skills')} placeholder="e.g. TypeScript, React, Node.js" className="mt-1" />
            </div>
            <div>
              <Label htmlFor="compensation_range">Compensation Range</Label>
              <Input id="compensation_range" {...form.register('compensation_range')} placeholder="e.g. $150k–$180k" className="mt-1" />
            </div>
            <div>
              <Label htmlFor="target_start_date">Target Start Date</Label>
              <Input id="target_start_date" type="date" {...form.register('target_start_date')} className="mt-1" />
            </div>
            <div className="flex gap-2 justify-end">
              <Button type="button" variant="outline" onClick={() => setIsEditOpen(false)}>Cancel</Button>
              <Button type="submit" disabled={patchMutation.isPending}>
                {patchMutation.isPending ? 'Saving…' : 'Update Requisition'}
              </Button>
            </div>
          </form>
        </DialogContent>
      </Dialog>
    </div>
  )
}
```
