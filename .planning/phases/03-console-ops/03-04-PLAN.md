**Create `apps/console/app/(shell)/ops/requisitions/[id]/_components/shortlist-table.tsx`:**

```typescript
'use client'

import { useState } from 'react'
import {
  useReactTable,
  getCoreRowModel,
  flexRender,
  createColumnHelper,
} from '@tanstack/react-table'
import type { ColumnDef } from '@tanstack/react-table'
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import { apiClient } from '@brynzit/api-client'
import type { components } from '@brynzit/api-client'
import { toast } from 'sonner'
import { Badge } from '@brynzit/ui/components/ui/badge'
import { Button } from '@brynzit/ui/components/ui/button'
import {
  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
} from '@brynzit/ui/components/ui/table'
import { CheckCircle, PauseCircle, XCircle } from 'lucide-react'

// CRITICAL: GET /api/v1/ops/requisitions/{req_id}/shortlist returns { [key: string]: unknown }
// Define local type from supply SubmissionResponse shape (verified api.d.ts line 3031)
type ShortlistEntry = {
  submission_id: string
  candidate_id: string
  candidate_name?: string | null
  current_title?: string | null
  fit_score?: number | null
  rank?: number | null
  status: string
}

interface ShortlistTableProps {
  reqId: string
  onCandidateSelect: (candidateId: string) => void
  onApprove: (candidateId: string) => void
}

function shortlistStatusBadgeClass(status: string): string {
  switch (status) {
    case 'approved': return 'bg-green-500/20 text-green-700 border-green-500/40'
    case 'hold': return 'bg-amber-500/20 text-amber-700 border-amber-500/40'
    case 'passed': return 'bg-muted text-muted-foreground border-border'
    default: return 'bg-primary/10 text-primary border-primary/20' // shortlisted
  }
}

export function ShortlistTable({ reqId, onCandidateSelect, onApprove }: ShortlistTableProps) {
  const queryClient = useQueryClient()

  const { data: entries = [], isLoading } = useQuery({
    queryKey: ['ops-shortlist', reqId],
    queryFn: async () => {
      const { data, error } = await apiClient.GET(
        '/api/v1/ops/requisitions/{req_id}/shortlist',
        { params: { path: { req_id: reqId } } },
      )
      if (error) throw error
      // Safe cast at boundary — backend returns JSONB { [key: string]: unknown }
      const raw = data as Record<string, unknown>
      return (raw['submissions'] as ShortlistEntry[] | undefined) ?? []
    },
  })

  // CRITICAL: Shortlist actions use SUPPLY domain PATCH — NOT ops endpoint
  const actionMutation = useMutation({
    mutationFn: ({ submissionId, status }: { submissionId: string; status: string }) =>
      apiClient.PATCH('/api/v1/supply/shortlists/{submission_id}', {
        params: { path: { submission_id: submissionId } },
        body: { status } satisfies components['schemas']['PatchSubmissionRequest'],
      }),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['ops-shortlist', reqId] })
    },
    onError: () => { toast.error('Failed to update — try again.') },
  })

  const handleAction = (entry: ShortlistEntry, newStatus: 'approved' | 'hold' | 'passed') => {
    const prevStatus = entry.status
    const name = entry.candidate_name ?? 'Candidate'
    const toastMsg =
      newStatus === 'approved' ? `Candidate ${name} approved.`
      : newStatus === 'hold' ? `Marked ${name} as hold.`
      : `Passed on ${name}.`

    // Optimistic + 5-second undo (same as Phase 2 approve pattern)
    const timeoutId = setTimeout(() => {
      actionMutation.mutate({ submissionId: entry.submission_id, status: newStatus })
      if (newStatus === 'approved') onApprove(entry.candidate_id)
    }, 300)

    toast(toastMsg, {
      duration: 5000,
      action: {
        label: 'Undo',
        onClick: () => {
          clearTimeout(timeoutId)
          queryClient.invalidateQueries({ queryKey: ['ops-shortlist', reqId] })
        },
      },
    })
  }

  const columnHelper = createColumnHelper<ShortlistEntry>()

  const columns: ColumnDef<ShortlistEntry, unknown>[] = [
    columnHelper.accessor('rank', {
      header: '#',
      cell: (info) => <span className="text-xs text-muted-foreground">{info.getValue() ?? '—'}</span>,
      size: 48,
    }),
    columnHelper.accessor('candidate_name', {
      header: 'Candidate',
      cell: (info) => (
        <button
          className="text-sm text-left hover:text-primary cursor-pointer"
          onClick={() => onCandidateSelect(info.row.original.candidate_id)}
        >
          {info.getValue() ?? 'Unknown'}
        </button>
      ),
    }),
    columnHelper.accessor('current_title', {
      header: 'Current Title',
      cell: (info) => (
        <span className="text-sm text-muted-foreground truncate max-w-[200px] block">
          {info.getValue() ?? '—'}
        </span>
      ),
    }),
    columnHelper.accessor('fit_score', {
      header: 'Fit Score',
      cell: (info) => {
        const score = info.getValue()
        return <span className="text-sm font-semibold">{score != null ? `${score}%` : '—'}</span>
      },
      size: 80,
    }),
    columnHelper.accessor('status', {
      header: 'Status',
      cell: (info) => (
        <Badge
          variant="outline"
          className={`text-xs ${shortlistStatusBadgeClass(info.getValue())}`}
        >
          {info.getValue()}
        </Badge>
      ),
      size: 120,
    }),
    columnHelper.display({
      id: 'actions',
      header: '',
      cell: (info) => {
        const entry = info.row.original
        return (
          <div className="flex items-center justify-end gap-1">
            <Button
              variant="ghost"
              size="icon"
              className="size-8"
              aria-label="Approve candidate"
              disabled={entry.status === 'approved' || entry.status === 'passed'}
              onClick={() => handleAction(entry, 'approved')}
            >
              <CheckCircle className="size-4" />
            </Button>
            <Button
              variant="ghost"
              size="icon"
              className="size-8"
              aria-label="Hold candidate"
              disabled={entry.status !== 'shortlisted'}
              onClick={() => handleAction(entry, 'hold')}
            >
              <PauseCircle className="size-4" />
            </Button>
            <Button
              variant="ghost"
              size="icon"
              className="size-8 hover:text-destructive"
              aria-label="Pass on candidate"
              disabled={entry.status === 'passed'}
              onClick={() => handleAction(entry, 'passed')}
            >
              <XCircle className="size-4" />
            </Button>
          </div>
        )
      },
      size: 120,
    }),
  ]

  const table = useReactTable({
    data: entries,
    columns,
    getCoreRowModel: getCoreRowModel(),
  })

  if (isLoading) return <p className="text-sm text-muted-foreground py-4 text-center">Loading shortlist…</p>

  if (entries.length === 0) {
    return (
      <p className="text-sm text-muted-foreground py-8 text-center">
        No candidates shortlisted yet. Trigger a match above to populate this list.
      </p>
    )
  }

  return (
    <div className="rounded-lg border border-border overflow-hidden">
      <Table>
        <TableHeader>
          {table.getHeaderGroups().map((hg) => (
            <TableRow key={hg.id}>
              {hg.headers.map((header) => (
                <TableHead key={header.id} style={{ width: header.column.getSize() || undefined }}>
                  {header.isPlaceholder ? null : flexRender(header.column.columnDef.header, header.getContext())}
                </TableHead>
              ))}
            </TableRow>
          ))}
        </TableHeader>
        <TableBody>
          {table.getRowModel().rows.map((row) => (
            <TableRow key={row.id} className="h-10 cursor-pointer hover:bg-muted/30">
              {row.getVisibleCells().map((cell) => (
                <TableCell key={cell.id}>
                  {flexRender(cell.column.columnDef.cell, cell.getContext())}
                </TableCell>
              ))}
            </TableRow>
          ))}
        </TableBody>
      </Table>
    </div>
  )
}
```

'use client'

import { useState } from 'react'
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import { apiClient } from '@brynzit/api-client'
import type { components } from '@brynzit/api-client'
import { toast } from 'sonner'
import {
  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
} from '@brynzit/ui/components/ui/select'
import { Badge } from '@brynzit/ui/components/ui/badge'
import { Button } from '@brynzit/ui/components/ui/button'
import { CalendarPlus } from 'lucide-react'

// Valid transitions per D-13 from CONTEXT.md
const VALID_TRANSITIONS: Record<string, string[]> = {
  scheduled: ['accepted', 'declined'],
  accepted: ['completed'],
  declined: [],   // terminal
  completed: [],  // terminal
}

interface InterviewsSectionProps {
  reqId: string
  approvedCandidateId: string | null
}

export function InterviewsSection({ reqId, approvedCandidateId }: InterviewsSectionProps) {
  const queryClient = useQueryClient()

  // NOTE: MSW-only — no live GET /api/v1/ops/interviews endpoint (api.d.ts get?: never line 548)
  const { data: interviews = [] } = useQuery({
    queryKey: ['ops-interviews', reqId],
    queryFn: async () => {
      const url = new URL(
        '/api/v1/ops/interviews',
        process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:8000',
      )
      url.searchParams.set('requisition_id', reqId)
      const res = await fetch(url.toString())
      if (!res.ok) throw new Error('Failed to fetch interviews')
      return res.json() as Promise<components['schemas']['InterviewResponse'][]>
    },
  })

  const patchMutation = useMutation({
    mutationFn: ({ interviewId, status }: { interviewId: string; status: string }) =>
      apiClient.PATCH('/api/v1/ops/interviews/{interview_id}', {
        params: { path: { interview_id: interviewId } },
        body: { status } satisfies components['schemas']['PatchInterviewRequest'],
      }),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['ops-interviews', reqId] })
    },
    onError: () => {
      // Pitfall 5: backend returns 409 on invalid state transitions
      toast.error('Invalid status transition.')
      queryClient.invalidateQueries({ queryKey: ['ops-interviews', reqId] })
    },
  })

  const createMutation = useMutation({
    mutationFn: (body: components['schemas']['CreateInterviewRequest']) =>
      apiClient.POST('/api/v1/ops/interviews', { body }),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['ops-interviews', reqId] })
      toast('Interview created.')
    },
    onError: () => { toast.error('Failed to create interview.') },
  })

  return (
    <div>
      {interviews.length === 0 ? (
        <p className="text-sm text-muted-foreground py-4 text-center">
          No interviews scheduled. Use &apos;Schedule Interview&apos; above to create one.
        </p>
      ) : (
        <div className="flex flex-col gap-2">
          {interviews.map((interview) => {
            const nextStates = VALID_TRANSITIONS[interview.status] ?? []
            const isTerminal = nextStates.length === 0

            return (
              <div
                key={interview.id}
                className="flex items-center gap-4 rounded-lg border border-border bg-card px-4 py-3 h-10"
              >
                <span className="flex-1 text-sm">{interview.candidate_id}</span>
                {interview.scheduled_at && (
                  <span className="text-xs text-muted-foreground w-[120px]">
                    {new Date(interview.scheduled_at).toLocaleDateString()}
                  </span>
                )}
                <Select
                  value={interview.status}
                  disabled={isTerminal}
                  onValueChange={(newStatus) =>
                    patchMutation.mutate({ interviewId: interview.id, status: newStatus })
                  }
                >
                  <SelectTrigger className="w-[160px]">
                    <SelectValue />
                  </SelectTrigger>
                  <SelectContent>
                    <SelectItem value={interview.status}>{interview.status}</SelectItem>
                    {nextStates.map((s) => (
                      <SelectItem key={s} value={s}>{s}</SelectItem>
                    ))}
                  </SelectContent>
                </Select>
                <InterviewStatusBadge status={interview.status} />
              </div>
            )
          })}
        </div>
      )}

      {approvedCandidateId && (
        <Button
          variant="outline"
          size="sm"
          className="mt-3"
          disabled={createMutation.isPending}
          onClick={() =>
            createMutation.mutate({
              requisition_id: reqId,
              candidate_id: approvedCandidateId,
            })
          }
        >
          <CalendarPlus className="mr-1 size-4" />Schedule Interview
        </Button>
      )}
    </div>
  )
}

function InterviewStatusBadge({ status }: { status: string }) {
  if (status === 'accepted' || status === 'completed') {
    return (
      <Badge
        variant="outline"
        className="text-xs bg-[oklch(0.45_0.15_145)]/20 text-[oklch(0.45_0.15_145)] border-[oklch(0.45_0.15_145)]/40"
      >
        {status}
      </Badge>
    )
  }
  if (status === 'declined') {
    return (
      <Badge variant="outline" className="text-xs border-amber-500 text-amber-600">
        {status}
      </Badge>
    )
  }
  return <Badge variant="secondary" className="text-xs">{status}</Badge>
}
