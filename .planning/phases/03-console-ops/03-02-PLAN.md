'use client'

import { useQuery } from '@tanstack/react-query'
import { apiClient } from '@brynzit/api-client'
import { ClientFields } from './_components/client-fields'
import { ClientNotes } from './_components/client-notes'
import { ClientRequisitions } from './_components/client-requisitions'

export default function ClientDetailPage({ params }: { params: { id: string } }) {
  const { data, isLoading, error } = useQuery({
    queryKey: ['ops-client', params.id],
    queryFn: async () => {
      const { data, error } = await apiClient.GET('/api/v1/ops/clients/{client_id}', {
        params: { path: { client_id: params.id } },
      })
      if (error) throw error
      return data
    },
  })

  if (isLoading) return <div className="p-6"><p className="text-muted-foreground text-sm">Loading…</p></div>
  if (error || !data) return <div className="p-6"><p className="text-destructive text-sm">Client not found.</p></div>

  return (
    <div className="flex h-[calc(100vh-3.5rem)]">
      {/* Left panel — client fields + compact req list */}

        <div>
          <h1 className="text-xl font-semibold">{data.name}</h1>
        </div>
        <ClientFields client={data} />
        <ClientRequisitions clientId={params.id} />
      </div>
      {/* Right panel — notes log */}
      <div className="flex flex-1 flex-col overflow-y-auto p-6">
        <ClientNotes clientId={params.id} />
      </div>
    </div>
  )
}

**Create `apps/console/app/(shell)/ops/clients/[id]/_components/client-notes.tsx`:**

This component MUST NOT use `dangerouslySetInnerHTML` anywhere — notes may contain pasted Slack/WhatsApp content (XSS vector).

```typescript
'use client'

import { useState } from 'react'
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import { apiClient } from '@brynzit/api-client'
import type { components } from '@brynzit/api-client'
import { toast } from 'sonner'
import { Textarea } from '@brynzit/ui/components/ui/textarea'
import { Button } from '@brynzit/ui/components/ui/button'

interface ClientNotesProps { clientId: string }

export function ClientNotes({ clientId }: ClientNotesProps) {
  const [noteText, setNoteText] = useState('')
  const queryClient = useQueryClient()

  const { data: notes } = useQuery({
    queryKey: ['client-notes', clientId],
    queryFn: async () => {
      const { data, error } = await apiClient.GET('/api/v1/ops/clients/{client_id}/notes', {
        params: { path: { client_id: clientId } },
      })
      if (error) throw error
      return data
    },
  })

  // Reverse-chronological sort
  const sortedNotes = [...(notes ?? [])].sort(
    (a, b) => new Date(b.created_at).getTime() - new Date(a.created_at).getTime(),
  )

  const addNoteMutation = useMutation({
    mutationFn: (text: string) =>
      apiClient.POST('/api/v1/ops/clients/{client_id}/notes', {
        params: { path: { client_id: clientId } },
        body: { text, channel: 'other' },
      }),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['client-notes', clientId] })
      setNoteText('')
    },
    onError: () => {
      toast.error('Failed to save note — try again.')
    },
  })

  return (
    <div className="flex flex-col h-full">
      <h2 className="text-base font-semibold mb-4">Notes</h2>

      {/* Notes log */}
      <div className="flex flex-col gap-3 mb-4 flex-1 overflow-y-auto">
        {sortedNotes.length === 0 ? (
          <p className="text-sm text-muted-foreground italic">
            No notes yet. Paste Slack or WhatsApp logs here.
          </p>
        ) : (
          sortedNotes.map((note) => (
            <div key={note.id} className="rounded-lg bg-muted/30 px-4 py-3">
              <p className="text-xs text-muted-foreground mb-1">
                {new Date(note.created_at).toLocaleString()}
              </p>
              {/* SECURITY: Render as text node — NEVER dangerouslySetInnerHTML */}
              <p className="text-sm whitespace-pre-wrap">{note.text}</p>
            </div>
          ))
        )}
      </div>

      {/* Note input — sticky at bottom */}
      <div className="border-t border-border pt-3">
        <Textarea
          value={noteText}
          onChange={(e) => setNoteText(e.target.value)}
          placeholder="Paste notes, Slack threads, or WhatsApp logs…"
          className="min-h-[80px] resize-none"
        />
        <Button
          className="mt-2 ml-auto block"
          size="sm"
          disabled={noteText.trim().length === 0 || addNoteMutation.isPending}
          onClick={() => addNoteMutation.mutate(noteText)}
        >
          {addNoteMutation.isPending ? 'Saving…' : 'Add Note'}
        </Button>
      </div>
    </div>
  )
}
```

```typescript
'use client'

import { useQuery } from '@tanstack/react-query'
import { apiClient } from '@brynzit/api-client'
import type { components } from '@brynzit/api-client'
import Link from 'next/link'
import { Plus } from 'lucide-react'
import { Badge } from '@brynzit/ui/components/ui/badge'
import { Separator } from '@brynzit/ui/components/ui/separator'

interface ClientRequisitionsProps { clientId: string }

export function ClientRequisitions({ clientId }: ClientRequisitionsProps) {
  const { data: requisitions } = useQuery({
    queryKey: ['client-requisitions', clientId],
    queryFn: async () => {
      const { data, error } = await apiClient.GET('/api/v1/ops/clients/{client_id}/requisitions', {
        params: { path: { client_id: clientId } },
      })
      if (error) throw error
      return data
    },
  })

  const reqs = requisitions ?? []
  const shown = reqs.slice(0, 5)
  const hasMore = reqs.length > 5

  return (
    <div>
      <Separator className="mb-4" />
      <div className="flex items-center justify-between mb-2">
        <h3 className="text-base font-semibold">Requisitions</h3>
        <Link
          href={`/ops/requisitions?client_id=${clientId}`}
          className="text-xs text-primary flex items-center gap-1"
        >
          <Plus className="size-3" />Create Requisition
        </Link>
      </div>

      {reqs.length === 0 ? (
        <p className="text-sm text-muted-foreground">No requisitions yet.</p>
      ) : (
        <div className="flex flex-col gap-1">
          {shown.map((req) => (
            <Link
              key={req.id}
              href={`/ops/requisitions/${req.id}`}
              className="flex items-center gap-2 rounded-md px-3 py-2 hover:bg-muted/50 transition-colors"
            >
              <span className="flex-1 text-sm truncate">{req.title}</span>
              <Badge variant="secondary" className="text-xs">{req.status}</Badge>
            </Link>
          ))}
          {hasMore && (
            <Link
              href={`/ops/requisitions?client_id=${clientId}`}
              className="text-xs text-primary mt-2"
            >
              View all requisitions for this client →
            </Link>
          )}
        </div>
      )}
    </div>
  )
}
```

describe('Client notes security (OPS-01)', () => {
  it('note text is NOT rendered via dangerouslySetInnerHTML', () => {
    // Test that the component pattern uses text node rendering
    // This is a pattern enforcement test — whitespace-pre-wrap p tag, not innerHTML
    const renderNote = (text: string) => `<p class="text-sm whitespace-pre-wrap">${text.replace(/</g, '&lt;').replace(/>/g, '&gt;')}</p>`
    const xssPayload = '<script>alert("xss")</script>'
    const rendered = renderNote(xssPayload)
    expect(rendered).not.toContain('<script>')
    expect(rendered).toContain('&lt;script&gt;')
  })

  it('notes are sorted reverse-chronological (newest first)', () => {
    const notes = [
      { id: '1', created_at: '2026-06-10T10:00:00.000Z', text: 'older note', client_id: 'c1', channel: 'other', author: null },
      { id: '2', created_at: '2026-06-14T10:00:00.000Z', text: 'newer note', client_id: 'c1', channel: 'other', author: null },
    ]
    const sorted = [...notes].sort(
      (a, b) => new Date(b.created_at).getTime() - new Date(a.created_at).getTime(),
    )
    expect(sorted[0].id).toBe('2')
    expect(sorted[1].id).toBe('1')
  })
})
